function bqf_hex_to_rgba( $hex, $alpha = 1 ) {
	$hex = ltrim( (string) $hex, '#' );

	if ( 3 === strlen( $hex ) ) {
		$hex = $hex[0] . $hex[0] . $hex[1] . $hex[1] . $hex[2] . $hex[2];
	}

	if ( 6 !== strlen( $hex ) || ! ctype_xdigit( $hex ) ) {
		$hex = '74041C';
	}

	return sprintf(
		'rgba(%d,%d,%d,%s)',
		hexdec( substr( $hex, 0, 2 ) ),
		hexdec( substr( $hex, 2, 2 ) ),
		hexdec( substr( $hex, 4, 2 ) ),
		rtrim( rtrim( number_format( (float) $alpha, 2, '.', '' ), '0' ), '.' )
	);
}

/**
 * Les extensions retenues pour un champ fichier, nettoyées.
 */
function bqf_field_extensions( $field ) {
	$list = array();

	foreach ( (array) $field['options'] as $line ) {
		foreach ( preg_split( '/[\s,;]+/', (string) $line ) as $ext ) {
			$ext = strtolower( ltrim( trim( (string) $ext ), '.' ) );

			if ( '' === $ext || ! preg_match( '/^[a-z0-9]{1,8}$/', $ext ) ) {
				continue;
			}

			if ( in_array( $ext, bqf_blocked_extensions(), true ) ) {
				continue;
			}

			$list[] = $ext;
		}
	}

	$list = array_values( array_unique( $list ) );

	return $list ? $list : bqf_default_extensions();
}

/**
 * Les types MIME correspondant à une liste d'extensions.
 */
function bqf_mimes_for( $extensions ) {
	$mimes = array();

	foreach ( wp_get_mime_types() as $pattern => $mime ) {
		foreach ( explode( '|', $pattern ) as $ext ) {
			if ( in_array( $ext, $extensions, true ) ) {
				$mimes[ $pattern ] = $mime;
				break;
			}
		}
	}

	return $mimes;
}

/**
 * Les champs d'un formulaire, nettoyés et prêts à l'emploi.
 */
function bqf_get_fields( $form_id ) {
	$fields = get_post_meta( (int) $form_id, '_bqf_fields', true );

	if ( ! is_array( $fields ) ) {
		return array();
	}

	$clean = array();
	$used  = array();

	foreach ( $fields as $field ) {
		if ( empty( $field['label'] ) ) {
			continue;
		}

		$type = isset( $field['type'] ) ? (string) $field['type'] : 'text';
		if ( ! array_key_exists( $type, bqf_field_types() ) ) {
			$type = 'text';
		}

		$key = sanitize_key( str_replace( '-', '_', sanitize_title( $field['label'] ) ) );
		if ( '' === $key ) {
			$key = 'champ';
		}

		$base = $key;
		$i    = 2;
		while ( in_array( $key, $used, true ) ) {
			$key = $base . '_' . $i;
			$i++;
		}
		$used[] = $key;

		$options = array();
		if ( ! empty( $field['options'] ) ) {
			foreach ( preg_split( '/\r\n|\r|\n/', (string) $field['options'] ) as $line ) {
				$line = trim( $line );
				if ( '' !== $line ) {
					$options[] = $line;
				}
			}
		}

		$profils = array();
		if ( ! empty( $field['profils'] ) ) {
			foreach ( (array) $field['profils'] as $profil ) {
				$profil = trim( (string) $profil );
				if ( '' !== $profil ) {
					$profils[] = $profil;
				}
			}
		}

		$clean[] = array(
			'key'         => $key,
			'label'       => (string) $field['label'],
			'type'        => $type,
			'profils'     => $profils,
			'width'       => in_array( ( isset( $field['width'] ) ? $field['width'] : '' ), array( 'half', 'third' ), true ) ? $field['width'] : 'full',
			'required'    => ! empty( $field['required'] ),
			'placeholder' => isset( $field['placeholder'] ) ? (string) $field['placeholder'] : '',
			'options'     => $options,
		);
	}

	return $clean;
}

/**
 * Le champ « sélecteur de profil » d'un formulaire, s'il existe.
 */
function bqf_profil_field( $fields ) {
	foreach ( (array) $fields as $field ) {
		if ( 'profil' === $field['type'] ) {
			return $field;
		}
	}

	return null;
}

/**
 * La liste des profils déclarés dans un formulaire.
 */
function bqf_form_profiles( $fields ) {
	$field = bqf_profil_field( $fields );

	return $field ? $field['options'] : array();
}

function bqf_get_profile_settings( $form_id ) {
	$saved = get_post_meta( (int) $form_id, '_bqf_profils', true );

}

/**
 * Les réglages d'un profil précis, complétés par ceux du formulaire.
 */
function bqf_profile_settings_for( $profile, $profiles, $settings ) {
	$row = isset( $profiles[ $profile ] ) ? $profiles[ $profile ] : array();

	$email  = isset( $row['email'] ) ? (string) $row['email'] : '';
	$help   = isset( $row['help'] ) ? (string) $row['help'] : '';
	$bouton = isset( $row['bouton'] ) ? (string) $row['bouton'] : '';

	return array(
		'email'  => is_email( $email ) ? $email : $settings['email'],
		'help'   => $help,
		'bouton' => '' !== trim( $bouton ) ? $bouton : $settings['bouton'],
	);
}

/**
 * Un champ est-il visible pour le profil actif ?
 */
function bqf_field_visible( $field, $profile ) {
	if ( empty( $field['profils'] ) ) {
		return true;
	}

	return in_array( $profile, $field['profils'], true );
}

/**
 * Les réglages d'un formulaire, avec leurs valeurs par défaut.
 */
function bqf_get_settings( $form_id ) {
	$form_id = (int) $form_id;

	$defaults = array(
		'email'   => BQF_MAIL,
		'sujet'   => 'Nouveau message depuis le site',
		'intro'   => '',
		'bouton'  => 'Envoyer',
		'succes'  => 'Merci, votre message a bien été envoyé. Nous revenons vers vous rapidement.',
		'rgpd'    => '',
		'mention' => '',
		'poids'   => '5',
	);

	$saved = array(
		'email'  => (string) get_post_meta( $form_id, '_bqf_email', true ),
		'sujet'  => (string) get_post_meta( $form_id, '_bqf_sujet', true ),
		'intro'  => (string) get_post_meta( $form_id, '_bqf_intro', true ),
		'bouton' => (string) get_post_meta( $form_id, '_bqf_bouton', true ),
		'succes' => (string) get_post_meta( $form_id, '_bqf_succes', true ),
		'rgpd'    => (string) get_post_meta( $form_id, '_bqf_rgpd', true ),
		'mention' => (string) get_post_meta( $form_id, '_bqf_mention', true ),
		'poids'   => (string) get_post_meta( $form_id, '_bqf_poids', true ),
	);

	foreach ( $saved as $key => $value ) {
		if ( '' === trim( $value ) ) {
			$saved[ $key ] = $defaults[ $key ];
		}
	}

	if ( ! is_email( $saved['email'] ) ) {
		$saved['email'] = BQF_MAIL;
	}

	$saved['poids'] = max( 1, min( 64, (int) $saved['poids'] ) );

	return $saved;
}

/**
 * Une ligne du constructeur de champs.
 */
function bqf_render_field_row( $index, $field, $profiles = array() ) {
	$field = wp_parse_args(
		(array) $field,
		array(
			'label'       => '',
			'type'        => 'text',
			'width'       => 'full',
			'required'    => 0,
			'placeholder' => '',
			'options'     => '',
			'profils'     => array(),
		)
	);

	$name     = 'bqf_fields[' . $index . ']';
	$has_opts  = in_array( $field['type'], bqf_types_with_list(), true );
	$is_file   = ( 'file' === $field['type'] );
	$is_profil = ( 'profil' === $field['type'] );
	$chosen    = (array) $field['profils'];

	$html  = '<div class="bqf-row" data-row>';

	$html .= '<div class="bqf-row__head">';
	$html .= '<span class="bqf-row__handle dashicons dashicons-menu-alt"></span>';
	$html .= '<span class="bqf-row__title" data-row-title>' . esc_html( $field['label'] ? $field['label'] : 'Nouveau champ' ) . '</span>';
	$html .= '<span class="bqf-row__actions">';
	$html .= '<button type="button" class="bqf-row__btn" data-move="up" title="Monter"><span class="dashicons dashicons-arrow-up-alt2"></span></button>';
	$html .= '<button type="button" class="bqf-row__btn" data-move="down" title="Descendre"><span class="dashicons dashicons-arrow-down-alt2"></span></button>';
	$html .= '<button type="button" class="bqf-row__btn bqf-row__btn--del" data-remove title="Supprimer"><span class="dashicons dashicons-trash"></span></button>';
	$html .= '</span>';
	$html .= '</div>';

	$html .= '<div class="bqf-row__body">';

	$html .= '<label class="bqf-row__field bqf-row__field--label">';
	$html .= '<span class="bqf-admin__label">Libellé</span>';
	$html .= '<input type="text" class="bqf-admin__input" name="' . esc_attr( $name ) . '[label]" value="' . esc_attr( $field['label'] ) . '" data-label placeholder="Organisation" />';
	$html .= '</label>';

	$html .= '<label class="bqf-row__field">';
	$html .= '<span class="bqf-admin__label">Type</span>';
	$html .= '<select class="bqf-admin__input" name="' . esc_attr( $name ) . '[type]" data-type>';
	foreach ( bqf_field_types() as $value => $caption ) {
		$html .= '<option value="' . esc_attr( $value ) . '"' . selected( $field['type'], $value, false ) . '>' . esc_html( $caption ) . '</option>';
	}
	$html .= '</select>';
	$html .= '</label>';

	$html .= '<label class="bqf-row__field">';
	$html .= '<span class="bqf-admin__label">Largeur</span>';
	$html .= '<select class="bqf-admin__input" name="' . esc_attr( $name ) . '[width]">';
	$html .= '<option value="full"' . selected( $field['width'], 'full', false ) . '>Pleine largeur — 1 champ par ligne</option>';
	$html .= '<option value="half"' . selected( $field['width'], 'half', false ) . '>Demie — 2 champs par ligne</option>';
	$html .= '<option value="third"' . selected( $field['width'], 'third', false ) . '>Tiers — 3 champs par ligne</option>';
	$html .= '</select>';
	$html .= '</label>';

	$html .= '<label class="bqf-row__field bqf-row__field--check">';
	$html .= '<input type="checkbox" name="' . esc_attr( $name ) . '[required]" value="1"' . checked( $field['required'], 1, false ) . ' />';
	$html .= '<span>Obligatoire</span>';
	$html .= '</label>';

	$html .= '<label class="bqf-row__field bqf-row__field--wide">';
	$html .= '<span class="bqf-admin__label">Texte d\'aide</span>';
	$html .= '<input type="text" class="bqf-admin__input" name="' . esc_attr( $name ) . '[placeholder]" value="' . esc_attr( $field['placeholder'] ) . '" placeholder="Affiché en gris dans le champ vide" />';
	$html .= '</label>';

	$html .= '<label class="bqf-row__field bqf-row__field--wide bqf-row__options"' . ( $has_opts ? '' : ' hidden' ) . ' data-options>';
	$html .= '<span class="bqf-admin__label" data-options-label>' . ( $is_file ? 'Extensions autorisées, séparées par une virgule' : 'Choix proposés, un par ligne' ) . '</span>';
	$html .= '<textarea class="bqf-admin__input bqf-admin__textarea" name="' . esc_attr( $name ) . '[options]" rows="4" placeholder="Entreprise&#10;Académique&#10;Institutionnel">' . esc_textarea( $field['options'] ) . '</textarea>';
	$html .= '<span class="bqf-admin__hint" data-options-hint' . ( $is_file ? '' : ' hidden' ) . '>Laissez vide pour accepter : ' . esc_html( implode( ', ', bqf_default_extensions() ) ) . '. Les fichiers exécutables sont toujours refusés.</span>';
	$html .= '</label>';

	if ( $profiles ) {
		$html .= '<div class="bqf-row__field bqf-row__field--wide bqf-row__profils"' . ( $is_profil ? ' hidden' : '' ) . ' data-profils-box>';
		$html .= '<span class="bqf-admin__label">Afficher uniquement pour</span>';
		$html .= '<span class="bqf-chips">';

		foreach ( $profiles as $profile ) {
			$on = in_array( $profile, $chosen, true );

			$html .= '<label class="bqf-chip' . ( $on ? ' is-on' : '' ) . '">';
			$html .= '<input type="checkbox" name="' . esc_attr( $name ) . '[profils][]" value="' . esc_attr( $profile ) . '"' . checked( $on, true, false ) . ' />';
			$html .= '<span>' . esc_html( $profile ) . '</span>';
			$html .= '</label>';
		}

		$html .= '</span>';
		$html .= '<span class="bqf-admin__hint">Aucune case cochée : le champ est affiché pour tous les profils. La liste se met à jour après enregistrement si vous modifiez les profils.</span>';
		$html .= '</div>';
	}

	$html .= '</div></div>';

	return $html;
}

/**
 * Encadré des profils : destinataire, phrase d'aide et bouton.
 */
function bqf_render_profils_box( $post ) {
	$fields   = bqf_get_fields( $post->ID );
	$profiles = bqf_form_profiles( $fields );
	$rows     = bqf_get_profile_settings( $post->ID );
	$settings = bqf_get_settings( $post->ID );

	if ( ! $profiles ) {
		echo '<div class="bqf-admin"><p class="bqf-admin__hint">'
			. 'Ce formulaire n\'a pas de sélecteur de profil. Ajoutez un champ de type '
			. '<strong>Sélecteur de profil (pastilles)</strong> et saisissez la liste des profils, un par ligne, '
			. 'puis enregistrez : le tableau des destinataires apparaîtra ici.'
			. '</p></div>';
		return;
	}

	$html  = '<div class="bqf-admin">';
	$html .= '<p class="bqf-admin__hint">Chaque profil peut être routé vers une adresse différente, avec sa propre phrase d\'aide et son propre libellé de bouton. Laissez vide pour reprendre les réglages généraux du formulaire.</p>';
	$html .= '<table class="bqf-profils"><thead><tr>';
	$html .= '<th>Profil</th><th>Adresse de réception</th><th>Phrase d\'aide</th><th>Libellé du bouton</th>';
	$html .= '</tr></thead><tbody>';

	foreach ( array_values( $profiles ) as $i => $profile ) {
		$row    = isset( $rows[ $profile ] ) ? $rows[ $profile ] : array();
		$name   = 'bqf_profils[' . $i . ']';
		$email  = isset( $row['email'] ) ? $row['email'] : '';
		$help   = isset( $row['help'] ) ? $row['help'] : '';
		$bouton = isset( $row['bouton'] ) ? $row['bouton'] : '';

		$html .= '<tr>';
		$html .= '<th scope="row">' . esc_html( $profile ) . '<input type="hidden" name="' . esc_attr( $name ) . '[name]" value="' . esc_attr( $profile ) . '" /></th>';
		$html .= '<td><input type="email" class="bqf-admin__input" name="' . esc_attr( $name ) . '[email]" value="' . esc_attr( $email ) . '" placeholder="' . esc_attr( $settings['email'] ) . '" /></td>';
		$html .= '<td><input type="text" class="bqf-admin__input" name="' . esc_attr( $name ) . '[help]" value="' . esc_attr( $help ) . '" placeholder="Votre message sera transmis à…" /></td>';
		$html .= '<td><input type="text" class="bqf-admin__input" name="' . esc_attr( $name ) . '[bouton]" value="' . esc_attr( $bouton ) . '" placeholder="' . esc_attr( $settings['bouton'] ) . '" /></td>';
		$html .= '</tr>';
	}

	$html .= '</tbody></table></div>';

	echo $html; // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
}

	echo $html; // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped

add_action( 'save_post_' . BQF_CPT, 'bqf_save_form', 10, 2 );
function bqf_save_form( $post_id, $post ) {
	if ( ! isset( $_POST['bqf_form_nonce'] ) || ! wp_verify_nonce( sanitize_key( wp_unslash( $_POST['bqf_form_nonce'] ) ), 'bqf_save_form' ) ) {
		return;
	}

	if ( defined( 'DOING_AUTOSAVE' ) && DOING_AUTOSAVE ) {
		return;
	}

	if ( ! current_user_can( 'edit_post', $post_id ) ) {
		return;
	}

	$email = isset( $_POST['bqf_email'] ) ? sanitize_email( wp_unslash( $_POST['bqf_email'] ) ) : '';
	update_post_meta( $post_id, '_bqf_email', is_email( $email ) ? $email : BQF_MAIL );

	$poids = isset( $_POST['bqf_poids'] ) ? absint( $_POST['bqf_poids'] ) : 5;
	update_post_meta( $post_id, '_bqf_poids', max( 1, min( 64, $poids ) ) );

	foreach ( array( 'sujet', 'intro', 'bouton', 'rgpd', 'mention' ) as $key ) {
		$value = isset( $_POST[ 'bqf_' . $key ] ) ? sanitize_text_field( wp_unslash( $_POST[ 'bqf_' . $key ] ) ) : '';
		update_post_meta( $post_id, '_bqf_' . $key, $value );
	}

	$succes = isset( $_POST['bqf_succes'] ) ? sanitize_textarea_field( wp_unslash( $_POST['bqf_succes'] ) ) : '';
	update_post_meta( $post_id, '_bqf_succes', $succes );

	$fields = array();

	if ( isset( $_POST['bqf_fields'] ) && is_array( $_POST['bqf_fields'] ) ) {
		$raw = wp_unslash( $_POST['bqf_fields'] ); // phpcs:ignore WordPress.Security.ValidatedSanitizedInput

		foreach ( $raw as $row ) {
			if ( ! is_array( $row ) || empty( $row['label'] ) ) {
				continue;
			}

			$type = isset( $row['type'] ) ? sanitize_key( $row['type'] ) : 'text';
			if ( ! array_key_exists( $type, bqf_field_types() ) ) {
				$type = 'text';
			}

			$profils = array();
			if ( ! empty( $row['profils'] ) && is_array( $row['profils'] ) ) {
				foreach ( $row['profils'] as $profil ) {
					$profil = sanitize_text_field( (string) $profil );
					if ( '' !== $profil ) {
						$profils[] = $profil;
					}
				}
			}

			$fields[] = array(
				'label'       => sanitize_text_field( $row['label'] ),
				'type'        => $type,
				'profils'     => $profils,
				'width'       => in_array( ( isset( $row['width'] ) ? $row['width'] : '' ), array( 'half', 'third' ), true ) ? $row['width'] : 'full',
				'required'    => empty( $row['required'] ) ? 0 : 1,
				'placeholder' => isset( $row['placeholder'] ) ? sanitize_text_field( $row['placeholder'] ) : '',
				'options'     => isset( $row['options'] ) ? sanitize_textarea_field( $row['options'] ) : '',
			);
		}
	}

	update_post_meta( $post_id, '_bqf_fields', $fields );

	$profils = array();

	if ( isset( $_POST['bqf_profils'] ) && is_array( $_POST['bqf_profils'] ) ) {
		$raw = wp_unslash( $_POST['bqf_profils'] ); // phpcs:ignore WordPress.Security.ValidatedSanitizedInput

		foreach ( $raw as $row ) {
			if ( ! is_array( $row ) || empty( $row['name'] ) ) {
				continue;
			}

			$name  = sanitize_text_field( $row['name'] );
			$email = isset( $row['email'] ) ? sanitize_email( $row['email'] ) : '';

			$profils[ $name ] = array(
				'email'  => is_email( $email ) ? $email : '',
				'help'   => isset( $row['help'] ) ? sanitize_text_field( $row['help'] ) : '',
				'bouton' => isset( $row['bouton'] ) ? sanitize_text_field( $row['bouton'] ) : '',
			);
		}
	}

	update_post_meta( $post_id, '_bqf_profils', $profils );
}

/* -------------------------------------------------------------------------
 * 4. Messages reçus et colonnes d'administration
 * ---------------------------------------------------------------------- */

function bqf_render_message_box( $post ) {
	$data = get_post_meta( $post->ID, '_bqf_data', true );
	$meta = get_post_meta( $post->ID, '_bqf_meta', true );

	if ( ! is_array( $data ) || ! $data ) {
		echo '<p>Message vide.</p>';
		return;
	}

	$html = '<table class="bqf-msg">';

	foreach ( $data as $line ) {
		$value = isset( $line['value'] ) ? (string) $line['value'] : '';

		$html .= '<tr>';
		$html .= '<th>' . esc_html( isset( $line['label'] ) ? $line['label'] : '' ) . '</th>';
		$html .= '<td>' . nl2br( esc_html( $value ) ) . '</td>';
		$html .= '</tr>';
	}

	$files = get_post_meta( $post->ID, '_bqf_files', true );

	if ( is_array( $files ) && $files ) {
		$links = array();

		foreach ( $files as $i => $file ) {
			$exists = ! empty( $file['path'] ) && is_file( $file['path'] );

			if ( ! $exists ) {
				$links[] = '<span class="bqf-file-gone">' . esc_html( $file['name'] ) . ' (fichier supprimé)</span>';
				continue;
			}

			$url = wp_nonce_url(
				admin_url( 'admin-post.php?action=bqf_download&msg=' . (int) $post->ID . '&f=' . (int) $i ),
				'bqf_download_' . (int) $post->ID . '_' . (int) $i
			);

			$links[] = '<a class="bqf-file-link" href="' . esc_url( $url ) . '">'
				. '<span class="dashicons dashicons-download"></span>'
				. esc_html( $file['name'] ) . ' <em>(' . esc_html( size_format( (int) $file['size'] ) ) . ')</em>'
				. '</a>';
		}

		$html .= '<tr><th>Pièces jointes</th><td>' . implode( '<br />', $links ) . '</td></tr>';
	}

	if ( is_array( $meta ) ) {
		$html .= '<tr><th>Page d\'origine</th><td>' . esc_html( isset( $meta['url'] ) ? $meta['url'] : '' ) . '</td></tr>';
		$html .= '<tr><th>Formulaire</th><td>' . esc_html( isset( $meta['form'] ) ? $meta['form'] : '' ) . '</td></tr>';

		if ( ! empty( $meta['profil'] ) ) {
			$html .= '<tr><th>Profil</th><td>' . esc_html( $meta['profil'] ) . '</td></tr>';
		}
	}

	$html .= '</table>';

	echo $html; // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
}

add_filter( 'manage_' . BQF_CPT . '_posts_columns', 'bqf_form_columns' );
function bqf_form_columns( $columns ) {
	$new = array();

	if ( isset( $columns['cb'] ) ) {
		$new['cb'] = $columns['cb'];
	}

	$new['title']         = 'Formulaire';
	$new['bqf_shortcode'] = 'Shortcode';
	$new['bqf_email']     = 'Réception';
	$new['bqf_count']     = 'Champs';
	$new['date']          = isset( $columns['date'] ) ? $columns['date'] : 'Date';

	return $new;
}

add_action( 'manage_' . BQF_CPT . '_posts_custom_column', 'bqf_form_column_content', 10, 2 );
function bqf_form_column_content( $column, $post_id ) {
	if ( 'bqf_shortcode' === $column ) {
		echo '<code class="bqf-code">[bqp_formulaire id="' . (int) $post_id . '"]</code>';
		return;
	}

	if ( 'bqf_email' === $column ) {
		$s = bqf_get_settings( $post_id );
		echo esc_html( $s['email'] );
		return;
	}

	if ( 'bqf_count' === $column ) {
		echo esc_html( count( bqf_get_fields( $post_id ) ) );
	}
}

add_filter( 'manage_' . BQF_MSG . '_posts_columns', 'bqf_msg_columns' );
function bqf_msg_columns( $columns ) {
	$new = array();

	if ( isset( $columns['cb'] ) ) {
		$new['cb'] = $columns['cb'];
	}

	$new['title']    = 'Message';
	$new['bqf_form'] = 'Formulaire';
	$new['bqf_from'] = 'Expéditeur';
	$new['date']     = 'Reçu le';

	return $new;
}

add_action( 'manage_' . BQF_MSG . '_posts_custom_column', 'bqf_msg_column_content', 10, 2 );
function bqf_msg_column_content( $column, $post_id ) {
	$meta = get_post_meta( $post_id, '_bqf_meta', true );
	$meta = is_array( $meta ) ? $meta : array();

	if ( 'bqf_form' === $column ) {
		echo esc_html( isset( $meta['form'] ) ? $meta['form'] : '—' );
		return;
	}

	if ( 'bqf_from' === $column ) {
		$from = isset( $meta['email'] ) ? $meta['email'] : '';

		if ( $from ) {
			echo '<a href="mailto:' . esc_attr( $from ) . '">' . esc_html( $from ) . '</a>';
		} else {
			echo '<span style="color:#b0b0b0;">—</span>';
		}
	}
}

/**
 * Supprimer un message supprime aussi ses pièces jointes.
 */
add_action( 'before_delete_post', 'bqf_delete_message_files' );
function bqf_delete_message_files( $post_id ) {
	if ( BQF_MSG !== get_post_type( $post_id ) ) {
		return;
	}

	$files = get_post_meta( $post_id, '_bqf_files', true );

	if ( ! is_array( $files ) ) {
		return;
	}

	$base = realpath( bqf_upload_path() );

	foreach ( $files as $file ) {
		$real = ! empty( $file['path'] ) ? realpath( $file['path'] ) : false;

		if ( $real && $base && 0 === strpos( $real, $base ) && is_file( $real ) ) {
			wp_delete_file( $real );
		}
	}
}

function bqf_admin_js() {
	return "
	(function(){
		function ready(fn){
			if ('loading' === document.readyState) { document.addEventListener('DOMContentLoaded', fn); }
			else { fn(); }
		}

		ready(function(){
			var list  = document.getElementById('bqf-rows');
			var tpl   = document.getElementById('bqf-tpl');
			var add   = document.getElementById('bqf-add');
			var empty = document.getElementById('bqf-empty');
			var index = 1000;

			var copy  = document.getElementById('bqf-sc-copy');
			var input = document.getElementById('bqf-sc-input');

			if (copy && input) {
				copy.addEventListener('click', function(){
					input.select();
					try { document.execCommand('copy'); } catch (e) {}
					var old = copy.textContent;
					copy.textContent = 'Copié';
					setTimeout(function(){ copy.textContent = old; }, 1600);
				});
			}

			if (!list || !tpl || !add) { return; }

			function refreshEmpty(){
				if (empty) { empty.hidden = (list.children.length > 0); }
			}

			function optionsToggle(row){
				var type  = row.querySelector('[data-type]');
				var opts  = row.querySelector('[data-options]');
				var label = row.querySelector('[data-options-label]');
				var hint  = row.querySelector('[data-options-hint]');

				if (!type || !opts) { return; }

				var value  = type.value;
				var isList = ('select' === value || 'radio' === value);
				var isFile = ('file' === value);

				opts.hidden = !(isList || isFile);

				if (label) {
					label.textContent = isFile
						? 'Extensions autorisées, séparées par une virgule'
						: 'Choix proposés, un par ligne';
				}

				if (hint) { hint.hidden = !isFile; }
			}

			add.addEventListener('click', function(){
				index++;
				var html = tpl.innerHTML.split('__INDEX__').join(String(index));
				var box  = document.createElement('div');
				box.innerHTML = html;
				var row = box.firstElementChild;
				list.appendChild(row);
				refreshEmpty();
				optionsToggle(row);
				var first = row.querySelector('[data-label]');
				if (first) { first.focus(); }
			});

			list.addEventListener('click', function(e){
				var btn = e.target.closest('button');
				if (!btn) { return; }

				var row = btn.closest('[data-row]');
				if (!row) { return; }

				if (btn.hasAttribute('data-remove')) {
					e.preventDefault();
					row.remove();
					refreshEmpty();
					return;
				}

				var move = btn.getAttribute('data-move');
				if ('up' === move) {
					e.preventDefault();
					if (row.previousElementSibling) { list.insertBefore(row, row.previousElementSibling); }
				} else if ('down' === move) {
					e.preventDefault();
					if (row.nextElementSibling) { list.insertBefore(row.nextElementSibling, row); }
				}
			});

			list.addEventListener('input', function(e){
				if (!e.target.hasAttribute('data-label')) { return; }
				var row   = e.target.closest('[data-row]');
				var title = row ? row.querySelector('[data-row-title]') : null;
				if (title) { title.textContent = e.target.value || 'Nouveau champ'; }
			});

			list.addEventListener('change', function(e){
				if (!e.target.hasAttribute('data-type')) { return; }
				var row = e.target.closest('[data-row]');
				if (row) { optionsToggle(row); }
			});

			Array.prototype.forEach.call(list.children, optionsToggle);
			refreshEmpty();
		});
	})();
	";
}

/**
 * Le dossier des pièces jointes, créé et fermé au public.
 */
function bqf_upload_path() {
	$uploads = wp_upload_dir();
	$dir     = trailingslashit( $uploads['basedir'] ) . 'bqp-formulaires';

	if ( ! is_dir( $dir ) ) {
		wp_mkdir_p( $dir );
	}

	// Les pièces jointes peuvent contenir des données personnelles :
	// le dossier est fermé à l'accès direct, le téléchargement passe
	// par l'administration.
	$htaccess = $dir . '/.htaccess';
	if ( ! file_exists( $htaccess ) ) {
		file_put_contents( $htaccess, "Order deny,allow\nDeny from all\n<IfModule mod_authz_core.c>\nRequire all denied\n</IfModule>\n" ); // phpcs:ignore WordPress.WP.AlternativeFunctions
	}

	$index = $dir . '/index.php';
	if ( ! file_exists( $index ) ) {
		file_put_contents( $index, '<' . '?php' . "\n// Silence.\n" ); // phpcs:ignore WordPress.WP.AlternativeFunctions
	}

	return $dir;
}

add_filter( 'upload_dir', 'bqf_filter_upload_dir' );
function bqf_filter_upload_dir( $dirs ) {
	if ( empty( $GLOBALS['bqf_uploading'] ) ) {
		return $dirs;
	}

	$dirs['path']   = trailingslashit( $dirs['basedir'] ) . 'bqp-formulaires';
	$dirs['url']    = trailingslashit( $dirs['baseurl'] ) . 'bqp-formulaires';
	$dirs['subdir'] = '/bqp-formulaires';

	return $dirs;
}

/**
 * Nom de fichier aléatoire : le nom d'origine n'est jamais exposé.
 */
function bqf_unique_filename( $dir, $name, $ext ) {
	return wp_generate_password( 24, false, false ) . strtolower( $ext );
}

/**
 * Vérifie puis enregistre un fichier déposé.
 * Renvoie array('error' => string) ou array('file' => array).
 */
function bqf_check_upload( $upload, $field, $max_bytes ) {
	$label = $field['label'];

	if ( ! isset( $upload['error'] ) || UPLOAD_ERR_NO_FILE === $upload['error'] ) {
		return '';
	}

	if ( UPLOAD_ERR_INI_SIZE === $upload['error'] || UPLOAD_ERR_FORM_SIZE === $upload['error'] ) {
		return sprintf( 'Le fichier déposé dans « %s » est trop volumineux.', $label );
	}

	if ( UPLOAD_ERR_OK !== $upload['error'] ) {
		return sprintf( 'Le fichier déposé dans « %s » n\'a pas pu être reçu. Merci de réessayer.', $label );
	}

	if ( ! empty( $upload['size'] ) && $upload['size'] > $max_bytes ) {
		return sprintf(
			'Le fichier déposé dans « %1$s » dépasse %2$s Mo.',
			$label,
			number_format_i18n( $max_bytes / 1048576, 0 )
		);
	}

	$extensions = bqf_field_extensions( $field );
	$ext        = strtolower( pathinfo( (string) $upload['name'], PATHINFO_EXTENSION ) );

	if ( '' === $ext || in_array( $ext, bqf_blocked_extensions(), true ) || ! in_array( $ext, $extensions, true ) ) {
		return sprintf(
			'Le fichier déposé dans « %1$s » n\'est pas d\'un format accepté (%2$s).',
			$label,
			strtoupper( implode( ', ', $extensions ) )
		);
	}

	$check = wp_check_filetype_and_ext( $upload['tmp_name'], $upload['name'], bqf_mimes_for( $extensions ) );

	if ( empty( $check['ext'] ) || empty( $check['type'] ) ) {
		return sprintf( 'Le contenu du fichier déposé dans « %s » ne correspond pas à son extension.', $label );
	}

	return '';
}

/**
 * Enregistre un fichier déjà contrôlé.
 * Renvoie array('error' => string) ou array('file' => array).
 */
function bqf_store_upload( $upload, $field, $max_bytes ) {
	$label = $field['label'];
	$error = bqf_check_upload( $upload, $field, $max_bytes );

	if ( '' !== $error ) {
		return array( 'error' => $error );
	}

	if ( ! isset( $upload['error'] ) || UPLOAD_ERR_NO_FILE === $upload['error'] ) {
		return array( 'error' => '' );
	}

	$mimes = bqf_mimes_for( bqf_field_extensions( $field ) );

	if ( ! function_exists( 'wp_handle_upload' ) ) {
		require_once ABSPATH . 'wp-admin/includes/file.php';
	}

	bqf_upload_path();

	$GLOBALS['bqf_uploading'] = true;

	$moved = wp_handle_upload(
		$upload,
		array(
			'test_form'                => false,
			'mimes'                    => $mimes,
			'unique_filename_callback' => 'bqf_unique_filename',
		)
	);

	$GLOBALS['bqf_uploading'] = false;

	if ( isset( $moved['error'] ) ) {
		return array( 'error' => sprintf( 'Le fichier déposé dans « %1$s » a été refusé : %2$s', $label, $moved['error'] ) );
	}

	return array(
		'error' => '',
		'file'  => array(
			'label' => $label,
			'name'  => sanitize_file_name( (string) $upload['name'] ),
			'path'  => $moved['file'],
			'size'  => (int) $upload['size'],
		),
	);
}

/**
 * Téléchargement d'une pièce jointe depuis l'administration.
 */
add_action( 'admin_post_bqf_download', 'bqf_download_file' );
function bqf_download_file() {
	if ( ! current_user_can( 'edit_posts' ) ) {
		wp_die( 'Accès refusé.' );
	}

	$message_id = isset( $_GET['msg'] ) ? absint( $_GET['msg'] ) : 0;
	$index      = isset( $_GET['f'] ) ? absint( $_GET['f'] ) : 0;

	check_admin_referer( 'bqf_download_' . $message_id . '_' . $index );

	$files = get_post_meta( $message_id, '_bqf_files', true );

	if ( ! is_array( $files ) || ! isset( $files[ $index ] ) ) {
		wp_die( 'Fichier introuvable.' );
	}

	$file = $files[ $index ];
	$path = isset( $file['path'] ) ? (string) $file['path'] : '';
	$real = $path ? realpath( $path ) : false;
	$base = realpath( bqf_upload_path() );

	if ( ! $real || ! $base || 0 !== strpos( $real, $base ) || ! is_file( $real ) ) {
		wp_die( 'Fichier introuvable.' );
	}

	$type = wp_check_filetype( $real );

	nocache_headers();
	header( 'Content-Type: ' . ( $type['type'] ? $type['type'] : 'application/octet-stream' ) );
	header( 'Content-Disposition: attachment; filename="' . sanitize_file_name( $file['name'] ) . '"' );
	header( 'Content-Length: ' . filesize( $real ) );
	header( 'X-Content-Type-Options: nosniff' );

	readfile( $real ); // phpcs:ignore WordPress.WP.AlternativeFunctions
	exit;
}

/**
 * Stocke l'état d'une soumission pour que le shortcode puisse l'afficher.
 */
function bqf_state( $form_id, $set = null ) {
	static $store = array();

	$form_id = (int) $form_id;

	if ( null !== $set ) {
		$store[ $form_id ] = $set;
	}

	return isset( $store[ $form_id ] ) ? $store[ $form_id ] : array( 'errors' => array(), 'values' => array() );
}

add_action( 'template_redirect', 'bqf_handle_submission' );
function bqf_handle_submission() {
	// Au-delà de post_max_size, PHP vide $_POST : sans ce garde-fou,
	// l'envoi échouerait en silence.
	if ( empty( $_POST ) && isset( $_SERVER['REQUEST_METHOD'] ) && 'POST' === $_SERVER['REQUEST_METHOD'] && ! empty( $_SERVER['CONTENT_LENGTH'] ) ) {
		$GLOBALS['bqf_too_big'] = true;
		return;
	}

	if ( empty( $_POST['bqf_form_id'] ) ) {
		return;
	}

	$form_id = absint( $_POST['bqf_form_id'] );
	$form    = $form_id ? get_post( $form_id ) : null;

	if ( ! $form || BQF_CPT !== $form->post_type ) {
		return;
	}

	$errors = array();
	$values = array();
	$lines  = array();

	// Session expirée (page servie depuis un cache trop ancien).
	$nonce = isset( $_POST['bqf_nonce'] ) ? sanitize_key( wp_unslash( $_POST['bqf_nonce'] ) ) : '';
	if ( ! wp_verify_nonce( $nonce, 'bqf_send_' . $form_id ) ) {
		bqf_state( $form_id, array( 'errors' => array( 'Votre session a expiré. Merci de recharger la page puis de renvoyer le formulaire.' ), 'values' => array() ) );
		return;
	}

	// Piège à robots : ce champ est invisible, il doit rester vide.
	if ( ! empty( $_POST['bqf_hp'] ) ) {
		bqf_state( $form_id, array( 'errors' => array( 'Envoi refusé.' ), 'values' => array() ) );
		return;
	}

	// Formulaire rempli trop vite pour être humain.
	$started = isset( $_POST['bqf_t'] ) ? absint( $_POST['bqf_t'] ) : 0;
	if ( $started && ( time() - $started ) < 4 ) {
		bqf_state( $form_id, array( 'errors' => array( 'Envoi refusé.' ), 'values' => array() ) );
		return;
	}

	$settings  = bqf_get_settings( $form_id );
	$fields    = bqf_get_fields( $form_id );

	$profiles       = bqf_form_profiles( $fields );
	$profil_field   = bqf_profil_field( $fields );
	$active_profile = '';

	if ( $profiles && $profil_field ) {
		$chosen = isset( $_POST[ 'bqf_' . $profil_field['key'] ] )
			? sanitize_text_field( wp_unslash( $_POST[ 'bqf_' . $profil_field['key'] ] ) )
			: '';

		$active_profile = in_array( $chosen, $profiles, true ) ? $chosen : $profiles[0];
	}

	$reply_to  = '';
	$sender    = '';
	$pending   = array();
	$max_bytes = (int) $settings['poids'] * 1048576;

	foreach ( $fields as $field ) {
		// Un champ réservé à un autre profil n'est ni exigé ni transmis.
		if ( ! bqf_field_visible( $field, $active_profile ) ) {
			continue;
		}

		$name = 'bqf_' . $field['key'];
		$raw  = isset( $_POST[ $name ] ) ? wp_unslash( $_POST[ $name ] ) : ''; // phpcs:ignore WordPress.Security.ValidatedSanitizedInput

		if ( 'file' === $field['type'] ) {
			$upload = isset( $_FILES[ $name ] ) ? $_FILES[ $name ] : null; // phpcs:ignore WordPress.Security.ValidatedSanitizedInput
			$empty  = ( ! $upload || ! isset( $upload['error'] ) || UPLOAD_ERR_NO_FILE === $upload['error'] );

			if ( $empty ) {
				if ( $field['required'] ) {
					$errors[] = sprintf( 'Le champ « %s » attend un fichier.', $field['label'] );
				}

				$values[ $field['key'] ] = '';
				$lines[]                 = array( 'label' => $field['label'], 'value' => '' );
				continue;
			}

			$problem = bqf_check_upload( $upload, $field, $max_bytes );

			if ( '' !== $problem ) {
				$errors[] = $problem;
			} else {
				$pending[] = array( 'upload' => $upload, 'field' => $field );
			}

			$values[ $field['key'] ] = '';
			$lines[]                 = array( 'label' => $field['label'], 'value' => sanitize_file_name( (string) $upload['name'] ) );
			continue;
		}

		switch ( $field['type'] ) {
			case 'email':
				$value = sanitize_email( (string) $raw );
				if ( '' !== $value && ! is_email( $value ) ) {
					$errors[] = sprintf( 'Le champ « %s » ne contient pas une adresse e-mail valide.', $field['label'] );
					$value    = sanitize_text_field( (string) $raw );
				} elseif ( $value && ! $reply_to ) {
					$reply_to = $value;
				}
				break;

			case 'url':
				$value = esc_url_raw( trim( (string) $raw ) );
				break;

			case 'number':
				$value = trim( (string) $raw );
				if ( '' !== $value && ! is_numeric( $value ) ) {
					$errors[] = sprintf( 'Le champ « %s » doit contenir un nombre.', $field['label'] );
				}
				break;

			case 'textarea':
				$value = sanitize_textarea_field( (string) $raw );
				break;

			case 'checkbox':
				$value = empty( $raw ) ? '' : 'Oui';
				break;

			case 'select':
			case 'radio':
				$value = sanitize_text_field( (string) $raw );
				if ( '' !== $value && $field['options'] && ! in_array( $value, $field['options'], true ) ) {
					$value = '';
				}
				break;

			case 'profil':
				$value = $active_profile;
				break;

			default:
				$value = sanitize_text_field( (string) $raw );
		}

		if ( $field['required'] && '' === trim( (string) $value ) ) {
			$errors[] = sprintf( 'Le champ « %s » est obligatoire.', $field['label'] );
		}

		if ( ! $sender && 'text' === $field['type'] && '' !== $value ) {
			$sender = $value;
		}

		$values[ $field['key'] ] = $value;
		$lines[]                 = array( 'label' => $field['label'], 'value' => $value );
	}

	// Case de consentement.
	if ( '' !== $settings['rgpd'] && empty( $_POST['bqf_rgpd'] ) ) {
		$errors[] = 'Merci de cocher la case de consentement.';
	}

	if ( $errors ) {
		bqf_state( $form_id, array( 'errors' => $errors, 'values' => $values ) );
		return;
	}

	// Les fichiers ne sont déplacés qu'une fois tout le reste validé.
	$stored      = array();
	$attachments = array();

	foreach ( $pending as $item ) {
		$result = bqf_store_upload( $item['upload'], $item['field'], $max_bytes );

		if ( ! empty( $result['error'] ) ) {
			$errors[] = $result['error'];
			continue;
		}

		if ( ! empty( $result['file'] ) ) {
			$stored[]      = $result['file'];
			$attachments[] = $result['file']['path'];
		}
	}

	if ( $errors ) {
		// On ne laisse jamais de fichier orphelin derrière une erreur.
		foreach ( $stored as $file ) {
			if ( ! empty( $file['path'] ) && is_file( $file['path'] ) ) {
				wp_delete_file( $file['path'] );
			}
		}

		bqf_state( $form_id, array( 'errors' => $errors, 'values' => $values ) );
		return;
	}

	$source = wp_get_referer();
	$source = $source ? $source : home_url( add_query_arg( array() ) );

	// Corps du message.
	$body = array( 'Nouveau message envoyé depuis ' . get_bloginfo( 'name' ) . '.', '' );

	foreach ( $lines as $line ) {
		$body[] = $line['label'] . ' : ' . ( '' === $line['value'] ? '—' : $line['value'] );
	}

	$body[] = '';
	$body[] = '---';
	$body[] = 'Formulaire : ' . get_the_title( $form_id );
	if ( $active_profile ) {
		$body[] = 'Profil : ' . $active_profile;
	}
	$body[] = 'Page : ' . $source;
	$body[] = 'Date : ' . wp_date( 'd/m/Y à H:i' );

	// L'expéditeur reste sur le domaine du site pour passer les filtres,
	// l'adresse du visiteur est placée en réponse.
	$host = wp_parse_url( home_url(), PHP_URL_HOST );
	$host = preg_replace( '/^www\./i', '', (string) $host );

	$headers = array(
		'Content-Type: text/plain; charset=UTF-8',
		sprintf( 'From: %s <%s>', wp_specialchars_decode( get_bloginfo( 'name' ), ENT_QUOTES ), 'no-reply@' . $host ),
	);

	if ( $reply_to ) {
		$headers[] = 'Reply-To: ' . $reply_to;
	}

	if ( $stored ) {
		$body[] = '';
		$body[] = count( $stored ) > 1 ? 'Fichiers joints :' : 'Fichier joint :';

		foreach ( $stored as $file ) {
			$body[] = '- ' . $file['label'] . ' : ' . $file['name'] . ' (' . size_format( $file['size'] ) . ')';
		}
	}

	$conf    = bqf_profile_settings_for( $active_profile, bqf_get_profile_settings( $form_id ), $settings );
	$to      = $profiles ? $conf['email'] : $settings['email'];
	$subject = $settings['sujet'] . ( $active_profile ? ' — ' . $active_profile : '' );

	$sent = wp_mail( $to, $subject, implode( "\n", $body ), $headers, $attachments );

	// Le message est archivé dans l'administration, même si l'e-mail échoue.
	$title = get_the_title( $form_id ) . ' — ' . ( $sender ? $sender : ( $reply_to ? $reply_to : 'sans nom' ) );

	$message_id = wp_insert_post(
		array(
			'post_type'   => BQF_MSG,
			'post_status' => 'publish',
			'post_title'  => wp_strip_all_tags( $title ),
		)
	);

	if ( $message_id && ! is_wp_error( $message_id ) ) {
		update_post_meta( $message_id, '_bqf_data', $lines );

		if ( $stored ) {
			update_post_meta( $message_id, '_bqf_files', $stored );
		}

		update_post_meta(
			$message_id,
			'_bqf_meta',
			array(
				'form'   => get_the_title( $form_id ),
				'email'  => $reply_to,
				'url'    => $source,
				'sent'   => $sent ? 'oui' : 'non',
				'profil' => $active_profile,
			)
		);
	}

	$redirect = remove_query_arg( 'bqf_ok' );
	$redirect = add_query_arg( 'bqf_ok', $form_id, $redirect );

	wp_safe_redirect( $redirect . '#bqf-' . $form_id );
	exit;
}

add_action( 'wp_enqueue_scripts', 'bqf_register_front_assets' );
function bqf_register_front_assets() {
	wp_register_style( 'bqf-formulaires', false, array(), BQF_VERSION );
	wp_add_inline_style( 'bqf-formulaires', bqf_front_css() );

	wp_register_script( 'bqf-formulaires', false, array(), BQF_VERSION, true );
	wp_add_inline_script( 'bqf-formulaires', bqf_front_js() );

	if ( is_singular() ) {
		$post = get_post();

		if ( $post instanceof WP_Post && has_shortcode( (string) $post->post_content, 'bqp_formulaire' ) ) {
			wp_enqueue_style( 'bqf-formulaires' );
			wp_enqueue_script( 'bqf-formulaires' );
		}
	}
}

add_shortcode( 'bqp_formulaire', 'bqf_shortcode' );
function bqf_shortcode( $atts ) {
	$atts = shortcode_atts(
		array(
			'id'       => '',
			'titre'    => '',
			'carte'    => 'oui',
			'colonnes' => '2',
		),
		$atts,
		'bqp_formulaire'
	);

	$form_id = absint( $atts['id'] );

	if ( ! $form_id ) {
		$found = get_posts(
			array(
				'post_type'        => BQF_CPT,
				'post_status'      => 'publish',
				'numberposts'      => 1,
				'orderby'          => 'ID',
				'order'            => 'ASC',
				'fields'           => 'ids',
				'suppress_filters' => false,
			)
		);

		$form_id = $found ? (int) $found[0] : 0;
	}

	$form = $form_id ? get_post( $form_id ) : null;

	if ( ! $form || BQF_CPT !== $form->post_type || 'publish' !== $form->post_status ) {
		if ( current_user_can( 'edit_posts' ) ) {
			return '<p class="bqf-notice">Formulaire introuvable. Vérifiez l\'identifiant du shortcode dans <strong>Formulaires</strong>.</p>';
		}

		return '';
	}

	$fields = bqf_get_fields( $form_id );

	if ( ! $fields ) {
		if ( current_user_can( 'edit_posts' ) ) {
			return '<p class="bqf-notice">Ce formulaire ne contient aucun champ. Ajoutez-en depuis <strong>Formulaires</strong>.</p>';
		}

		return '';
	}

	wp_enqueue_style( 'bqf-formulaires' );
	wp_enqueue_script( 'bqf-formulaires' );

	$settings = bqf_get_settings( $form_id );
	$state    = bqf_state( $form_id );
	$errors   = $state['errors'];
	$values   = $state['values'];

	$profiles       = bqf_form_profiles( $fields );
	$profile_rows   = bqf_get_profile_settings( $form_id );
	$profil_field   = bqf_profil_field( $fields );
	$active_profile = '';
	$profile_map    = array();

	if ( $profiles ) {
		$submitted = ( $profil_field && isset( $values[ $profil_field['key'] ] ) ) ? (string) $values[ $profil_field['key'] ] : '';

		$active_profile = in_array( $submitted, $profiles, true ) ? $submitted : $profiles[0];

		foreach ( $profiles as $profile ) {
			$conf = bqf_profile_settings_for( $profile, $profile_rows, $settings );

			$profile_map[ $profile ] = array(
				'help'   => $conf['help'],
				'bouton' => $conf['bouton'],
			);
		}
	}

	$active_conf = $profiles
		? bqf_profile_settings_for( $active_profile, $profile_rows, $settings )
		: array( 'email' => $settings['email'], 'help' => '', 'bouton' => $settings['bouton'] );

	if ( ! empty( $GLOBALS['bqf_too_big'] ) ) {
		$errors[] = 'L\'envoi est trop volumineux pour le serveur. Merci de déposer un fichier plus léger.';
	}

	$single = ( '1' === (string) $atts['colonnes'] );
	$card  = ! in_array( strtolower( $atts['carte'] ), array( 'non', 'no', 'false', '0' ), true );
	$anchor = 'bqf-' . $form_id;

	$title = get_the_title( $form_id );
	if ( '' !== trim( $atts['titre'] ) ) {
		$title = in_array( strtolower( $atts['titre'] ), array( 'non', 'no', 'false', '0' ), true ) ? '' : $atts['titre'];
	}

	$sent = isset( $_GET['bqf_ok'] ) && absint( $_GET['bqf_ok'] ) === $form_id; // phpcs:ignore WordPress.Security.NonceVerification

	$classes = 'bqf-form' . ( $card ? ' bqf-form--card' : '' ) . ( $single ? ' bqf-form--single' : '' );

	$data = '';
	if ( $profiles ) {
		$data = ' data-bqf-profils="' . esc_attr( wp_json_encode( $profile_map ) ) . '"';
	}

	$html  = '<section class="' . esc_attr( $classes ) . '" id="' . esc_attr( $anchor ) . '"' . $data . '>';

	if ( '' !== $title ) {
		$html .= '<h2 class="bqf-form__title">' . esc_html( $title ) . '</h2>';
	}

	if ( '' !== $settings['intro'] ) {
		$html .= '<p class="bqf-form__intro">' . esc_html( $settings['intro'] ) . '</p>';
	}

	if ( $profiles && '' !== $active_conf['help'] ) {
		$html .= '<p class="bqf-form__profil-help" data-profil-help>' . esc_html( $active_conf['help'] ) . '</p>';
	}

	if ( $sent ) {
		$html .= '<div class="bqf-success" role="status">';
		$html .= '<span class="bqf-success__icon" aria-hidden="true">';
		$html .= '<svg viewBox="0 0 24 24" width="22" height="22"><path d="M4 12.5l5 5L20 6.5" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round" stroke-linejoin="round"/></svg>';
		$html .= '</span>';
		$html .= '<p>' . esc_html( $settings['succes'] ) . '</p>';
		$html .= '</div></section>';

		return $html;
	}

	if ( $errors ) {
		$html .= '<div class="bqf-errors" role="alert"><ul>';
		foreach ( $errors as $error ) {
			$html .= '<li>' . esc_html( $error ) . '</li>';
		}
		$html .= '</ul></div>';
	}

	$has_file = false;
	foreach ( $fields as $field ) {
		if ( 'file' === $field['type'] ) {
			$has_file = true;
			break;
		}
	}

	$html .= '<form class="bqf-form__grid" method="post" action="' . esc_url( remove_query_arg( 'bqf_ok' ) ) . '#' . esc_attr( $anchor ) . '"' . ( $has_file ? ' enctype="multipart/form-data"' : '' ) . '>';

	foreach ( $fields as $field ) {
		$name  = 'bqf_' . $field['key'];
		$id    = $anchor . '-' . $field['key'];
		$value = isset( $values[ $field['key'] ] ) ? (string) $values[ $field['key'] ] : '';
		$req   = $field['required'] ? ' required' : '';
		$span  = ' bqf-field--' . $field['width'];

		$group = '';
		if ( ! empty( $field['profils'] ) ) {
			$group = ' data-for="' . esc_attr( implode( '|', $field['profils'] ) ) . '"'
				. ( bqf_field_visible( $field, $active_profile ) ? '' : ' hidden' );
		}

		// Le sélecteur de profil : une rangée de pastilles.
		if ( 'profil' === $field['type'] ) {
			if ( ! $field['options'] ) {
				continue;
			}

			$html .= '<div class="bqf-field bqf-field--full bqf-field--profils">';
			$html .= '<span class="bqf-field__label">' . esc_html( $field['label'] ) . '</span>';
			$html .= '<div class="bqf-pills" role="group" aria-label="' . esc_attr( $field['label'] ) . '">';

			foreach ( $field['options'] as $option ) {
				$on = ( $option === $active_profile );

				$html .= '<button type="button" class="bqf-pill' . ( $on ? ' is-active' : '' ) . '" data-profil="' . esc_attr( $option ) . '" aria-pressed="' . ( $on ? 'true' : 'false' ) . '">'
					. esc_html( $option ) . '</button>';
			}

			$html .= '</div>';
			$html .= '<input type="hidden" name="' . esc_attr( $name ) . '" value="' . esc_attr( $active_profile ) . '" data-profil-value />';
			$html .= '</div>';
			continue;
		}

		if ( 'checkbox' === $field['type'] ) {
			$html .= '<div class="bqf-field bqf-field--checkbox' . $span . '"' . $group . '>';
			$html .= '<label class="bqf-check">';
			$html .= '<input type="checkbox" name="' . esc_attr( $name ) . '" value="1"' . checked( $value, 'Oui', false ) . $req . ' />';
			$html .= '<span>' . esc_html( $field['label'] ) . ( $field['required'] ? ' <em class="bqf-req">*</em>' : '' ) . '</span>';
			$html .= '</label></div>';
			continue;
		}

		$html .= '<div class="bqf-field' . $span . '"' . $group . '>';

		$caption = esc_html( $field['label'] ) . ( $field['required'] ? ' <em class="bqf-req">*</em>' : '' );

		if ( 'radio' === $field['type'] ) {
			$html .= '<span class="bqf-field__label">' . $caption . '</span>';
		} else {
			$html .= '<label class="bqf-field__label" for="' . esc_attr( $id ) . '">' . $caption . '</label>';
		}

		if ( 'textarea' === $field['type'] ) {
			$html .= '<textarea class="bqf-input bqf-input--area" name="' . esc_attr( $name ) . '" id="' . esc_attr( $id ) . '" rows="4" placeholder="' . esc_attr( $field['placeholder'] ) . '"' . $req . '>' . esc_textarea( $value ) . '</textarea>';

		} elseif ( 'select' === $field['type'] ) {
			$empty = $field['placeholder'] ? $field['placeholder'] : 'Choisir';
			$html .= '<select class="bqf-input bqf-input--select" name="' . esc_attr( $name ) . '" id="' . esc_attr( $id ) . '"' . $req . '>';
			$html .= '<option value="">' . esc_html( $empty ) . '</option>';
			foreach ( $field['options'] as $option ) {
				$html .= '<option value="' . esc_attr( $option ) . '"' . selected( $value, $option, false ) . '>' . esc_html( $option ) . '</option>';
			}
			$html .= '</select>';

		} elseif ( 'radio' === $field['type'] ) {
			$html .= '<div class="bqf-radios">';
			foreach ( $field['options'] as $i => $option ) {
				$html .= '<label class="bqf-check">';
				$html .= '<input type="radio" name="' . esc_attr( $name ) . '" value="' . esc_attr( $option ) . '"' . checked( $value, $option, false ) . ( ( $field['required'] && 0 === $i ) ? ' required' : '' ) . ' />';
				$html .= '<span>' . esc_html( $option ) . '</span>';
				$html .= '</label>';
			}
			$html .= '</div>';

		} elseif ( 'file' === $field['type'] ) {
			$exts   = bqf_field_extensions( $field );
			$accept = '.' . implode( ',.', $exts );

			$html .= '<div class="bqf-file">';
			$html .= '<input type="file" class="bqf-file__input" name="' . esc_attr( $name ) . '" id="' . esc_attr( $id ) . '" accept="' . esc_attr( $accept ) . '"' . $req . ' />';
			$html .= '<label class="bqf-file__button" for="' . esc_attr( $id ) . '">';
			$html .= '<svg viewBox="0 0 20 20" width="16" height="16" aria-hidden="true" focusable="false"><path d="M10 14V4m0 0L6 8m4-4l4 4M3 15v1.5A1.5 1.5 0 004.5 18h11a1.5 1.5 0 001.5-1.5V15" fill="none" stroke="currentColor" stroke-width="1.6" stroke-linecap="round" stroke-linejoin="round"/></svg>';
			$html .= '<span>' . esc_html( $field['placeholder'] ? $field['placeholder'] : 'Choisir un fichier' ) . '</span>';
			$html .= '</label>';
			$html .= '<span class="bqf-file__name" data-file-name>Aucun fichier sélectionné</span>';
			$html .= '</div>';
			$html .= '<span class="bqf-help">' . esc_html( strtoupper( implode( ', ', $exts ) ) ) . ' — ' . esc_html( $settings['poids'] ) . ' Mo maximum.</span>';

		} else {
			$type  = in_array( $field['type'], array( 'email', 'tel', 'url', 'number', 'date' ), true ) ? $field['type'] : 'text';
			$html .= '<input type="' . esc_attr( $type ) . '" class="bqf-input" name="' . esc_attr( $name ) . '" id="' . esc_attr( $id ) . '" value="' . esc_attr( $value ) . '" placeholder="' . esc_attr( $field['placeholder'] ) . '"' . $req . ' />';
		}

		$html .= '</div>';
	}

	if ( '' !== $settings['rgpd'] ) {
		$html .= '<div class="bqf-field bqf-field--full bqf-field--checkbox">';
		$html .= '<label class="bqf-check"><input type="checkbox" name="bqf_rgpd" value="1" required /><span>' . esc_html( $settings['rgpd'] ) . ' <em class="bqf-req">*</em></span></label>';
		$html .= '</div>';
	}

	// Champ piège, invisible pour les visiteurs.
	$html .= '<div class="bqf-hp" aria-hidden="true"><label>Ne pas remplir<input type="text" name="bqf_hp" value="" tabindex="-1" autocomplete="off" /></label></div>';

	$html .= '<input type="hidden" name="bqf_form_id" value="' . (int) $form_id . '" />';
	$html .= '<input type="hidden" name="bqf_t" value="' . (int) time() . '" />';
	$html .= wp_nonce_field( 'bqf_send_' . $form_id, 'bqf_nonce', false, false );

	if ( '' !== $settings['mention'] ) {
		$html .= '<p class="bqf-form__mention bqf-field--full">' . esc_html( $settings['mention'] ) . '</p>';
	}

	$html .= '<div class="bqf-field bqf-field--full">';
	$html .= '<button type="submit" class="bqf-submit" data-profil-button>' . esc_html( $active_conf['bouton'] ) . '</button>';
	$html .= '</div>';

	$html .= '</form></section>';

	return $html;
}

/**
 * Affiche le nom du fichier choisi à côté du bouton.
 */
function bqf_front_js() {
	return "
	(function(){
		function initProfils(section){
			var map = {};
			try { map = JSON.parse(section.getAttribute('data-bqf-profils') || '{}'); } catch (e) { map = {}; }

			var pills  = section.querySelectorAll('.bqf-pill');
			var hidden = section.querySelector('[data-profil-value]');
			var help   = section.querySelector('[data-profil-help]');
			var button = section.querySelector('[data-profil-button]');
			var groups = section.querySelectorAll('[data-for]');

			if (!pills.length) { return; }

			function apply(profile){
				Array.prototype.forEach.call(pills, function(pill){
					var on = (pill.getAttribute('data-profil') === profile);
					if (on) { pill.classList.add('is-active'); } else { pill.classList.remove('is-active'); }
					pill.setAttribute('aria-pressed', on ? 'true' : 'false');
				});

				if (hidden) { hidden.value = profile; }

				Array.prototype.forEach.call(groups, function(group){
					var list = (group.getAttribute('data-for') || '').split('|');
					var on   = (list.indexOf(profile) !== -1);

					group.hidden = !on;

					// Un champ obligatoire mais masqué bloquerait l'envoi :
					// on le neutralise, il n'est alors pas transmis non plus.
					Array.prototype.forEach.call(group.querySelectorAll('input, select, textarea'), function(el){
						el.disabled = !on;
					});
				});

				var conf = map[profile] || {};

				if (help) {
					help.textContent = conf.help || '';
					help.hidden = !conf.help;
				}

				if (button && conf.bouton) { button.textContent = conf.bouton; }
			}

			Array.prototype.forEach.call(pills, function(pill){
				pill.addEventListener('click', function(){
					apply(pill.getAttribute('data-profil'));
				});
			});

			apply((hidden && hidden.value) ? hidden.value : pills[0].getAttribute('data-profil'));
		}

		function boot(){
			Array.prototype.forEach.call(document.querySelectorAll('.bqf-form[data-bqf-profils]'), initProfils);

			var inputs = document.querySelectorAll('.bqf-file__input');

			Array.prototype.forEach.call(inputs, function(input){
				input.addEventListener('change', function(){
					var box = input.parentNode.querySelector('[data-file-name]');
					if (!box) { return; }

					if (input.files && input.files.length) {
						box.textContent = input.files[0].name;
						box.classList.add('is-set');
					} else {
						box.textContent = 'Aucun fichier sélectionné';
						box.classList.remove('is-set');
					}
				});
			});
		}

		if ('loading' === document.readyState) {
			document.addEventListener('DOMContentLoaded', boot);
		} else {
			boot();
		}
	})();
	";
}
