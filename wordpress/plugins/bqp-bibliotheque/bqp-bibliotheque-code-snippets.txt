function bqb_hex_to_rgba( $hex, $alpha = 1 ) {
	$hex = ltrim( (string) $hex, '#' );

	if ( 3 === strlen( $hex ) ) {
		$hex = $hex[0] . $hex[0] . $hex[1] . $hex[1] . $hex[2] . $hex[2];
	}

	if ( 6 !== strlen( $hex ) || ! ctype_xdigit( $hex ) ) {
		$hex = '74041C';
	}

	return sprintf(
		'rgba(%d,%d,%d,%s)',
		hexdec( substr( $hex, 0, 2 ) ),
		hexdec( substr( $hex, 2, 2 ) ),
		hexdec( substr( $hex, 4, 2 ) ),
		rtrim( rtrim( number_format( (float) $alpha, 2, '.', '' ), '0' ), '.' )
	);
}

/**
 * Le type Document étant public, ses adresses doivent être enregistrées
 * une fois. Code Snippets n'a pas de hook d'activation : on le fait ici,
 * une seule fois par version.
 */
add_action( 'init', 'bqb_maybe_flush_rewrite', 99 );
function bqb_maybe_flush_rewrite() {
	if ( get_option( 'bqb_rewrite_version' ) === BQB_VERSION ) {
		return;
	}

	flush_rewrite_rules( false );
	update_option( 'bqb_rewrite_version', BQB_VERSION, false );
}

add_filter( 'enter_title_here', 'bqb_title_placeholder', 10, 2 );
function bqb_title_placeholder( $text, $post ) {
	if ( $post && BQB_CPT === $post->post_type ) {
		return 'Titre exact du document';
	}

	return $text;
}

/**
 * Libellé au-dessus de l'éditeur : c'est la notice éditoriale.
 */
add_action( 'edit_form_after_title', 'bqb_editor_heading' );
function bqb_editor_heading( $post ) {
	if ( BQB_CPT !== $post->post_type ) {
		return;
	}

	echo '<div class="bqb-editor-head">'
		. '<span class="bqb-admin__label">Contexte et notice éditoriale</span>'
		. '<span class="bqb-admin__hint">Présentation du document, contexte de publication, intérêt pour le lecteur. Facultatif, mais c\'est ce texte qui donne de la valeur à la page aux yeux de Google.</span>'
		. '</div>';
}

/**
 * Crée un niveau de l'arbre, puis ses enfants.
 */
function bqb_seed_tree( $taxonomy, $terms, $parent = 0 ) {
	foreach ( $terms as $term ) {
		$slug   = sanitize_title( $term['name'] );
		$exists = term_exists( $slug, $taxonomy, $parent );

		if ( $exists ) {
			$term_id = (int) ( is_array( $exists ) ? $exists['term_id'] : $exists );
		} else {
			$created = wp_insert_term(
				$term['name'],
				$taxonomy,
				array(
					'slug'        => $slug,
					'parent'      => (int) $parent,
					'description' => $term['description'],
				)
			);

			if ( is_wp_error( $created ) ) {
				continue;
			}

			$term_id = (int) $created['term_id'];

			foreach ( $term['meta'] as $key => $value ) {
				update_term_meta( $term_id, $key, $value );
			}
		}

		if ( $term['children'] ) {
			bqb_seed_tree( $taxonomy, $term['children'], $term_id );
		}
	}
}

add_action( 'admin_init', 'bqb_maybe_seed' );
function bqb_maybe_seed() {
	if ( get_option( 'bqb_seeded' ) ) {
		return;
	}

	if ( ! current_user_can( 'manage_categories' ) ) {
		return;
	}

	foreach ( bqb_seed_data() as $taxonomy => $terms ) {
		if ( taxonomy_exists( $taxonomy ) ) {
			bqb_seed_tree( $taxonomy, $terms );
		}
	}

	update_option( 'bqb_seeded', BQB_VERSION, false );
}

/**
 * Le champ lui-même, identique à l'ajout et à la modification.
 */
function bqb_term_field_input( $field, $value ) {
	list( $key, $type ) = $field;

	switch ( $type ) {
		case 'number':
			return '<input type="number" name="' . esc_attr( $key ) . '" id="' . esc_attr( $key ) . '" value="' . esc_attr( $value ) . '" step="1" style="width:120px;" />';

		case 'color':
			return '<input type="color" name="' . esc_attr( $key ) . '" id="' . esc_attr( $key ) . '" value="' . esc_attr( $value ? $value : '#74041C' ) . '" />';

		case 'url':
			return '<input type="url" name="' . esc_attr( $key ) . '" id="' . esc_attr( $key ) . '" value="' . esc_attr( $value ) . '" placeholder="https://" />';

		case 'textarea':
			return '<textarea name="' . esc_attr( $key ) . '" id="' . esc_attr( $key ) . '" rows="5">' . esc_textarea( $value ) . '</textarea>';

		case 'roles':
			$value = is_array( $value ) ? $value : array();
			$html  = '<span class="bqb-checks">';
			foreach ( bqb_person_roles() as $role => $label ) {
				$html .= '<label><input type="checkbox" name="' . esc_attr( $key ) . '[]" value="' . esc_attr( $role ) . '"' . checked( in_array( $role, $value, true ), true, false ) . ' /> ' . esc_html( $label ) . '</label>';
			}
			return $html . '</span>';

		case 'orgtype':
			$html = '<select name="' . esc_attr( $key ) . '" id="' . esc_attr( $key ) . '"><option value="">Choisir</option>';
			foreach ( bqb_org_types() as $slug => $label ) {
				$html .= '<option value="' . esc_attr( $slug ) . '"' . selected( $value, $slug, false ) . '>' . esc_html( $label ) . '</option>';
			}
			return $html . '</select>';

		case 'prix':
			$terms = get_terms( array( 'taxonomy' => 'bqb_prix', 'hide_empty' => false ) );
			$html  = '<select name="' . esc_attr( $key ) . '" id="' . esc_attr( $key ) . '"><option value="">Aucun</option>';
			if ( ! is_wp_error( $terms ) ) {
				foreach ( $terms as $term ) {
					$html .= '<option value="' . (int) $term->term_id . '"' . selected( (int) $value, (int) $term->term_id, false ) . '>' . esc_html( $term->name ) . '</option>';
				}
			}
			return $html . '</select>';

		case 'image':
			$src  = $value ? wp_get_attachment_image_url( (int) $value, 'thumbnail' ) : '';
			$html = '<span class="bqb-media" data-media>';
			$html .= '<img src="' . esc_url( $src ) . '" alt="" data-media-preview' . ( $src ? '' : ' hidden' ) . ' />';
			$html .= '<input type="hidden" name="' . esc_attr( $key ) . '" value="' . esc_attr( $value ) . '" data-media-id />';
			$html .= '<button type="button" class="button" data-media-select>' . ( $src ? 'Remplacer' : 'Choisir une image' ) . '</button> ';
			$html .= '<button type="button" class="button-link bqb-remove" data-media-remove' . ( $src ? '' : ' hidden' ) . '>Retirer</button>';
			return $html . '</span>';
	}

	return '';
}

function bqb_term_add_fields( $taxonomy ) {
	$fields = bqb_term_fields();

	if ( empty( $fields[ $taxonomy ] ) ) {
		return;
	}

	foreach ( $fields[ $taxonomy ] as $field ) {
		echo '<div class="form-field bqb-term-field">'
			. '<label for="' . esc_attr( $field[0] ) . '">' . esc_html( $field[2] ) . '</label>'
			. bqb_term_field_input( $field, '' ) // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
			. ( $field[3] ? '<p>' . esc_html( $field[3] ) . '</p>' : '' )
			. '</div>';
	}

	wp_nonce_field( 'bqb_term_meta', 'bqb_term_nonce' );
}

function bqb_term_edit_fields( $term, $taxonomy ) {
	$fields = bqb_term_fields();

	if ( empty( $fields[ $taxonomy ] ) ) {
		return;
	}

	foreach ( $fields[ $taxonomy ] as $field ) {
		$value = get_term_meta( $term->term_id, $field[0], true );

		echo '<tr class="form-field bqb-term-field">'
			. '<th scope="row"><label for="' . esc_attr( $field[0] ) . '">' . esc_html( $field[2] ) . '</label></th>'
			. '<td>' . bqb_term_field_input( $field, $value ) // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
			. ( $field[3] ? '<p class="description">' . esc_html( $field[3] ) . '</p>' : '' )
			. '</td></tr>';
	}

	wp_nonce_field( 'bqb_term_meta', 'bqb_term_nonce' );
}

function bqb_save_term_fields( $term_id, $tt_id = 0, $taxonomy = '' ) {
	if ( ! isset( $_POST['bqb_term_nonce'] ) || ! wp_verify_nonce( sanitize_key( wp_unslash( $_POST['bqb_term_nonce'] ) ), 'bqb_term_meta' ) ) {
		return;
	}

	if ( ! current_user_can( 'manage_categories' ) ) {
		return;
	}

	$term = get_term( $term_id );
	if ( ! $term || is_wp_error( $term ) ) {
		return;
	}

	$fields = bqb_term_fields();
	if ( empty( $fields[ $term->taxonomy ] ) ) {
		return;
	}

	foreach ( $fields[ $term->taxonomy ] as $field ) {
		list( $key, $type ) = $field;
		$raw = isset( $_POST[ $key ] ) ? wp_unslash( $_POST[ $key ] ) : ''; // phpcs:ignore WordPress.Security.ValidatedSanitizedInput

		switch ( $type ) {
			case 'number':
			case 'image':
			case 'prix':
				$value = ( '' === $raw ) ? '' : (int) $raw;
				break;
			case 'color':
				$value = sanitize_hex_color( (string) $raw );
				break;
			case 'url':
				$value = esc_url_raw( trim( (string) $raw ) );
				break;
			case 'textarea':
				$value = sanitize_textarea_field( (string) $raw );
				break;
			case 'roles':
				$value = array_values( array_intersect( array_map( 'sanitize_key', (array) $raw ), array_keys( bqb_person_roles() ) ) );
				break;
			case 'orgtype':
				$value = array_key_exists( (string) $raw, bqb_org_types() ) ? (string) $raw : '';
				break;
			default:
				$value = sanitize_text_field( (string) $raw );
		}

		if ( '' === $value || array() === $value || null === $value ) {
			delete_term_meta( $term_id, $key );
		} else {
			update_term_meta( $term_id, $key, $value );
		}
	}
}

add_action( 'init', 'bqb_hook_term_fields', 20 );
function bqb_hook_term_fields() {
	foreach ( array_keys( bqb_term_fields() ) as $taxonomy ) {
		add_action( $taxonomy . '_add_form_fields', 'bqb_term_add_fields' );
		add_action( $taxonomy . '_edit_form_fields', 'bqb_term_edit_fields', 10, 2 );
		add_action( 'created_' . $taxonomy, 'bqb_save_term_fields', 10, 3 );
		add_action( 'edited_' . $taxonomy, 'bqb_save_term_fields', 10, 3 );
	}
}

/**
 * « PDF · 6,4 Mo » à partir d'une pièce de la médiathèque.
 */
function bqb_file_label( $file_id ) {
	$path = get_attached_file( (int) $file_id );

	if ( ! $path ) {
		return '';
	}

	$ext  = strtoupper( pathinfo( $path, PATHINFO_EXTENSION ) );
	$size = file_exists( $path ) ? size_format( filesize( $path ), 1 ) : '';

	return trim( $ext . ( $size ? ' · ' . $size : '' ) );
}

/**
 * Recherche de documents pour le champ « Documents associés ».
 */
add_action( 'wp_ajax_bqb_search_documents', 'bqb_ajax_search_documents' );
function bqb_ajax_search_documents() {
	check_ajax_referer( 'bqb_related', 'nonce' );

	if ( ! current_user_can( 'edit_posts' ) ) {
		wp_send_json_error();
	}

	$term    = isset( $_GET['q'] ) ? sanitize_text_field( wp_unslash( $_GET['q'] ) ) : '';
	$exclude = isset( $_GET['exclude'] ) ? absint( $_GET['exclude'] ) : 0;

	if ( strlen( $term ) < 2 ) {
		wp_send_json_success( array() );
	}

	$posts = get_posts(
		array(
			'post_type'        => BQB_CPT,
			'post_status'      => array( 'publish', 'draft', 'pending', 'private', 'future' ),
			's'                => $term,
			'numberposts'      => 12,
			'exclude'          => $exclude ? array( $exclude ) : array(),
			'suppress_filters' => false,
		)
	);

	$out = array();
	foreach ( $posts as $p ) {
		$out[] = array(
			'id'    => (int) $p->ID,
			'title' => html_entity_decode( get_the_title( $p ), ENT_QUOTES, 'UTF-8' ),
			'year'  => (string) get_post_meta( $p->ID, '_bqb_annee', true ),
		);
	}

	wp_send_json_success( $out );
}

add_action( 'save_post_' . BQB_CPT, 'bqb_save_document', 10, 2 );
function bqb_save_document( $post_id, $post ) {
	if ( ! isset( $_POST['bqb_document_nonce'] ) || ! wp_verify_nonce( sanitize_key( wp_unslash( $_POST['bqb_document_nonce'] ) ), 'bqb_save_document' ) ) {
		return;
	}

	if ( defined( 'DOING_AUTOSAVE' ) && DOING_AUTOSAVE ) {
		return;
	}

	if ( wp_is_post_revision( $post_id ) || ! current_user_can( 'edit_post', $post_id ) ) {
		return;
	}

	$int = function ( $key, $min, $max ) {
		if ( ! isset( $_POST[ $key ] ) || '' === $_POST[ $key ] ) { // phpcs:ignore WordPress.Security.NonceVerification
			return '';
		}
		$n = (int) $_POST[ $key ]; // phpcs:ignore WordPress.Security.NonceVerification
		return ( $n >= $min && $n <= $max ) ? $n : '';
	};

	$annee = $int( 'bqb_annee', 1000, 2100 );
	$mois  = $int( 'bqb_mois', 1, 12 );
	$jour  = $int( 'bqb_jour', 1, 31 );

	// Un jour sans mois n'a pas de sens.
	if ( ! $mois ) {
		$jour = '';
	}

	$values = array(
		'_bqb_annee'     => $annee,
		'_bqb_mois'      => $mois,
		'_bqb_jour'      => $jour,
		'_bqb_edition'   => $int( 'bqb_edition', 1900, 2100 ),
		'_bqb_fichier'   => isset( $_POST['bqb_fichier'] ) ? absint( $_POST['bqb_fichier'] ) : '',
		'_bqb_lien'      => isset( $_POST['bqb_lien'] ) ? esc_url_raw( trim( wp_unslash( $_POST['bqb_lien'] ) ) ) : '',
		'_bqb_reference' => isset( $_POST['bqb_reference'] ) ? sanitize_textarea_field( wp_unslash( $_POST['bqb_reference'] ) ) : '',
		'_bqb_droits'    => isset( $_POST['bqb_droits'] ) ? sanitize_text_field( wp_unslash( $_POST['bqb_droits'] ) ) : '',
	);

	$acces                = isset( $_POST['bqb_acces'] ) ? sanitize_key( wp_unslash( $_POST['bqb_acces'] ) ) : 'public';
	$values['_bqb_acces'] = array_key_exists( $acces, bqb_access_levels() ) ? $acces : 'public';

	// Date de tri : toujours complète, pour trier et grouper par année.
	$values['_bqb_date_tri'] = $annee
		? sprintf( '%04d-%02d-%02d', $annee, $mois ? $mois : 1, $jour ? $jour : 1 )
		: '';

	foreach ( $values as $key => $value ) {
		if ( '' === $value || 0 === $value ) {
			delete_post_meta( $post_id, $key );
		} else {
			update_post_meta( $post_id, $key, $value );
		}
	}

	if ( isset( $_POST['bqb_associes_present'] ) ) {
		$related = isset( $_POST['bqb_associes'] ) ? array_map( 'absint', (array) wp_unslash( $_POST['bqb_associes'] ) ) : array();
		$related = array_values( array_unique( array_filter( $related, function ( $id ) use ( $post_id ) {
			return $id && $id !== (int) $post_id && BQB_CPT === get_post_type( $id );
		} ) ) );

		if ( $related ) {
			update_post_meta( $post_id, '_bqb_associes', $related );
		} else {
			delete_post_meta( $post_id, '_bqb_associes' );
		}
	}
}

add_filter( 'manage_' . BQB_CPT . '_posts_columns', 'bqb_document_columns' );
function bqb_document_columns( $columns ) {
	$new = array();

	if ( isset( $columns['cb'] ) ) {
		$new['cb'] = $columns['cb'];
	}

	$new['bqb_cover'] = '';
	$new['title']     = 'Document';

	foreach ( array( 'taxonomy-bqb_collection', 'taxonomy-bqb_nature' ) as $key ) {
		if ( isset( $columns[ $key ] ) ) {
			$new[ $key ] = $columns[ $key ];
		}
	}

	$new['bqb_annee']  = 'Année';
	$new['bqb_acces']  = 'Accès';
	$new['bqb_resume'] = 'Résumé';
	$new['bqb_source'] = 'Fichier ou lien';
	$new['date']       = isset( $columns['date'] ) ? $columns['date'] : 'Date';

	return $new;
}

add_action( 'manage_' . BQB_CPT . '_posts_custom_column', 'bqb_document_column_content', 10, 2 );
function bqb_document_column_content( $column, $post_id ) {
	switch ( $column ) {
		case 'bqb_cover':
			if ( has_post_thumbnail( $post_id ) ) {
				echo get_the_post_thumbnail( $post_id, array( 44, 60 ), array( 'class' => 'bqb-col-cover' ) ); // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
			} else {
				echo '<span class="bqb-col-cover bqb-col-cover--empty dashicons dashicons-book-alt"></span>';
			}
			break;

		case 'bqb_annee':
			$year = get_post_meta( $post_id, '_bqb_annee', true );
			echo $year ? (int) $year : '<span class="bqb-muted">—</span>';
			break;

		case 'bqb_acces':
			$acces = get_post_meta( $post_id, '_bqb_acces', true );
			$acces = $acces ? $acces : 'public';
			$label = array(
				'public'      => 'Public',
				'adherents'   => 'Adhérents',
				'partenaires' => 'Partenaires',
				'sur-place'   => 'Sur place',
			);
			echo '<span class="bqb-badge bqb-badge--' . esc_attr( $acces ) . '">' . esc_html( isset( $label[ $acces ] ) ? $label[ $acces ] : 'Public' ) . '</span>';
			break;

		case 'bqb_resume':
			$excerpt = get_post_field( 'post_excerpt', $post_id );
			echo '' !== trim( (string) $excerpt )
				? '<span class="bqb-ok dashicons dashicons-yes-alt" title="Résumé renseigné"></span>'
				: '<span class="bqb-missing">Manquant</span>';
			break;

		case 'bqb_source':
			$file = (int) get_post_meta( $post_id, '_bqb_fichier', true );
			$link = get_post_meta( $post_id, '_bqb_lien', true );
			$out  = array();
			if ( $file ) {
				$out[] = esc_html( bqb_file_label( $file ) );
			}
			if ( $link ) {
				$out[] = '<a href="' . esc_url( $link ) . '" target="_blank" rel="noopener noreferrer">Lien</a>';
			}
			echo $out ? implode( ' · ', $out ) : '<span class="bqb-muted">—</span>'; // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
			break;
	}
}

add_action( 'pre_get_posts', 'bqb_document_admin_query' );
function bqb_document_admin_query( $query ) {
	if ( ! is_admin() || ! $query->is_main_query() || BQB_CPT !== $query->get( 'post_type' ) ) {
		return;
	}

	if ( 'bqb_annee' === $query->get( 'orderby' ) ) {
		// Les documents sans date restent dans la liste.
		$query->set(
			'meta_query',
			array(
				'relation' => 'OR',
				'bqb_date' => array( 'key' => '_bqb_date_tri', 'compare' => 'EXISTS' ),
				array( 'key' => '_bqb_date_tri', 'compare' => 'NOT EXISTS' ),
			)
		);
		$query->set( 'orderby', 'bqb_date' );
	}
}

/**
 * Filtres par collection, nature et thème au-dessus de la liste :
 * indispensable dès qu'il y a plusieurs centaines de documents.
 */
add_action( 'restrict_manage_posts', 'bqb_document_filters' );
function bqb_document_filters( $post_type ) {
	if ( BQB_CPT !== $post_type ) {
		return;
	}

	$filters = array(
		'bqb_collection' => 'Toutes les collections',
		'bqb_nature'     => 'Toutes les natures',
		'bqb_theme'      => 'Tous les thèmes',
	);

	foreach ( $filters as $taxonomy => $all ) {
		$current = isset( $_GET[ $taxonomy ] ) ? sanitize_title( wp_unslash( $_GET[ $taxonomy ] ) ) : ''; // phpcs:ignore WordPress.Security.NonceVerification

		wp_dropdown_categories(
			array(
				'taxonomy'        => $taxonomy,
				'name'            => $taxonomy,
				'value_field'     => 'slug',
				'selected'        => $current,
				'show_option_all' => $all,
				'hierarchical'    => true,
				'hide_empty'      => false,
				'orderby'         => 'name',
			)
		);
	}
}

	if ( ! $screen ) {
		return;

	$ours = ( BQB_CPT === $screen->post_type )
		|| ( $screen->taxonomy && array_key_exists( $screen->taxonomy, bqb_taxonomies() ) )
		|| ( false !== strpos( (string) $hook, 'bqb-' ) );

	if ( ! $ours ) {
		return;
	}

	wp_enqueue_style( 'bqb-fonts', 'https://fonts.googleapis.com/css2?family=Cormorant+Garamond:wght@500;600;700&family=Inter:wght@400;500;600;700&display=swap', array(), null );

		wp_enqueue_media();
	}

	wp_register_style( 'bqb-admin', false, array(), BQB_VERSION );
	wp_enqueue_style( 'bqb-admin' );
	wp_add_inline_style( 'bqb-admin', bqb_admin_css() . bqb_admin_css_v2() );

				input.addEventListener('input', function(){
					clearTimeout(timer);
					var q = input.value.trim();
					if (q.length < 2) { results.hidden = true; return; }

					timer = setTimeout(function(){
						var url = BQB.ajax + '?action=bqb_search_documents&nonce=' + encodeURIComponent(BQB.nonce)
							+ '&exclude=' + encodeURIComponent(current) + '&q=' + encodeURIComponent(q);

	";

/**
 * L'adresse d'un terme : sa page dédiée si elle existe, sinon sa page
 * d'archive, sinon la page catalogue filtrée.
 */
function bqb_term_url( $term, $args = array() ) {
	if ( ! $term instanceof WP_Term ) {
		return '';
	}

	$page = (int) get_term_meta( $term->term_id, 'bqb_page', true );
	$tax  = get_taxonomy( $term->taxonomy );

	if ( $page && 'publish' === get_post_status( $page ) ) {
		$url = get_permalink( $page );
	} elseif ( $tax && $tax->public ) {
		$url = get_term_link( $term );
	} else {
		$settings = bqb_settings();
		$url      = $settings['page_catalogue'] ? get_permalink( (int) $settings['page_catalogue'] ) : '';

		if ( ! $url ) {
			return '';
		}

		$args = array_merge( array( 'f_' . bqb_short_name( $term->taxonomy ) => $term->slug ), $args );
	}

	if ( is_wp_error( $url ) || ! $url ) {
		return '';
	}

	return $args ? add_query_arg( array_map( 'rawurlencode', $args ), $url ) : $url;
}

/**
 * Nombre de documents publiés pour une combinaison de critères.
 * $criteria : array( 'collection' => array( 'slug' ), 'nature' => … ).
 */
function bqb_count( $criteria ) {
	$version = (string) get_option( 'bqb_cache_version', '1' );
	$key     = 'bqb_c_' . md5( $version . wp_json_encode( $criteria ) );
	$cached  = get_transient( $key );

	if ( false !== $cached ) {
		return (int) $cached;
	}

	$args                   = bqb_query_args( $criteria );
	$args['posts_per_page'] = 1;
	$args['fields']         = 'ids';
	$args['no_found_rows']  = false;

	$query = new WP_Query( $args );
	$count = (int) $query->found_posts;

	set_transient( $key, $count, DAY_IN_SECONDS );

	return $count;
}

/**
 * Transforme des critères en arguments WP_Query.
 * Plusieurs valeurs dans une même famille : l'une OU l'autre.
 * Plusieurs familles : l'une ET l'autre.
 */
function bqb_query_args( $criteria ) {
	$args = array(
		'post_type'           => BQB_CPT,
		'post_status'         => 'publish',
		'ignore_sticky_posts' => true,
		'tax_query'           => array( 'relation' => 'AND' ),
		'meta_query'          => array( 'relation' => 'AND' ),
	);

	foreach ( bqb_short_map() as $short => $taxonomy ) {
		if ( empty( $criteria[ $short ] ) ) {
			continue;
		}

		$slugs = array_filter( array_map( 'sanitize_title', (array) $criteria[ $short ] ) );

		if ( $slugs ) {
			$args['tax_query'][] = array(
				'taxonomy'         => $taxonomy,
				'field'            => 'slug',
				'terms'            => $slugs,
				'include_children' => true,
			);
		}
	}

	if ( ! empty( $criteria['and'] ) ) {
		foreach ( (array) $criteria['and'] as $short => $slugs ) {
			$map   = bqb_short_map();
			$slugs = array_filter( array_map( 'sanitize_title', (array) $slugs ) );

			if ( isset( $map[ $short ] ) && $slugs ) {
				$args['tax_query'][] = array(
					'taxonomy'         => $map[ $short ],
					'field'            => 'slug',
					'terms'            => $slugs,
					'include_children' => true,
				);
			}
		}
	}

	if ( ! empty( $criteria['q'] ) ) {
		$args['s'] = sanitize_text_field( $criteria['q'] );
	}

	if ( ! empty( $criteria['annee'] ) ) {
		$range = array_map( 'intval', explode( '-', (string) $criteria['annee'] ) );

		if ( 2 === count( $range ) && $range[0] && $range[1] ) {
			$args['meta_query'][] = array( 'key' => '_bqb_annee', 'value' => array( min( $range ), max( $range ) ), 'type' => 'NUMERIC', 'compare' => 'BETWEEN' );
		} elseif ( $range[0] ) {
			$args['meta_query'][] = array( 'key' => '_bqb_annee', 'value' => $range[0], 'type' => 'NUMERIC', 'compare' => '=' );
		}
	}

	if ( ! empty( $criteria['acces'] ) && array_key_exists( $criteria['acces'], bqb_access_levels() ) ) {
		if ( 'public' === $criteria['acces'] ) {
			$args['meta_query'][] = array(
				'relation' => 'OR',
				array( 'key' => '_bqb_acces', 'value' => 'public' ),
				array( 'key' => '_bqb_acces', 'compare' => 'NOT EXISTS' ),
			);
		} else {
			$args['meta_query'][] = array( 'key' => '_bqb_acces', 'value' => $criteria['acces'] );
		}
	}

	$tri = isset( $criteria['tri'] ) ? $criteria['tri'] : 'recent';

	if ( 'titre' === $tri ) {
		$args['orderby'] = array( 'title' => 'ASC' );
	} else {
		// Les documents sans date restent listés, après les autres.
		$args['meta_query']['bqb_date'] = array(
			'relation'     => 'OR',
			'bqb_date_ok'  => array( 'key' => '_bqb_date_tri', 'compare' => 'EXISTS' ),
			'bqb_date_non' => array( 'key' => '_bqb_date_tri', 'compare' => 'NOT EXISTS' ),
		);
		$args['orderby'] = array(
			'bqb_date_ok' => ( 'ancien' === $tri ) ? 'ASC' : 'DESC',
			'title'       => 'ASC',
		);
	}

	if ( 1 === count( $args['tax_query'] ) ) {
		unset( $args['tax_query'] );
	}

	return $args;
}

/**
 * Tous les termes, rangés : ordre choisi, puis ordre alphabétique.
 * Renvoie array( parent_id => array( WP_Term, … ) ).
 */
function bqb_term_children_map( $taxonomy ) {
	$terms = get_terms( array( 'taxonomy' => $taxonomy, 'hide_empty' => false ) );
	$map   = array();

	if ( is_wp_error( $terms ) ) {
		return $map;
	}

	foreach ( $terms as $term ) {
		$term->bqb_ordre              = get_term_meta( $term->term_id, 'bqb_ordre', true );
		$map[ (int) $term->parent ][] = $term;
	}

	foreach ( $map as $parent => $list ) {
		usort(
			$map[ $parent ],
			function ( $a, $b ) {
				$oa = ( '' === $a->bqb_ordre ) ? 9999 : (int) $a->bqb_ordre;
				$ob = ( '' === $b->bqb_ordre ) ? 9999 : (int) $b->bqb_ordre;

				return ( $oa === $ob ) ? strcasecmp( remove_accents( $a->name ), remove_accents( $b->name ) ) : ( $oa - $ob );
			}
		);
	}

	return $map;
}

function bqb_pill( $taxonomy, $term, $checked, $extra_class = '' ) {
	return '<label class="bqb-pill' . ( $checked ? ' is-on' : '' ) . ( $extra_class ? ' ' . $extra_class : '' ) . '" data-name="' . esc_attr( remove_accents( strtolower( $term->name ) ) ) . '">'
		. '<input type="checkbox" name="bqb_terms[' . esc_attr( $taxonomy ) . '][]" value="' . (int) $term->term_id . '"' . checked( $checked, true, false ) . ' />'
		. '<span>' . esc_html( $term->name ) . '</span>'
		. '</label>';

function bqb_render_classement_box( $post ) {
	wp_nonce_field( 'bqb_save_classement', 'bqb_classement_nonce' );

	$taxonomies = bqb_taxonomies();
	$html       = '<div class="bqb-admin bqb-class">';

	foreach ( bqb_picker_config() as $taxonomy => $conf ) {
		list( $mode, $help ) = $conf;

		$assigned = wp_get_object_terms( $post->ID, $taxonomy, array( 'fields' => 'ids' ) );
		$assigned = is_wp_error( $assigned ) ? array() : array_map( 'intval', $assigned );
		$label    = $taxonomies[ $taxonomy ][0];
		$wide     = in_array( $mode, array( 'tree', 'tokens' ), true ) || 'bqb_nature' === $taxonomy;

		$html .= '<section class="bqb-class__block' . ( $wide ? ' is-wide' : '' ) . '" data-tax="' . esc_attr( $taxonomy ) . '" data-mode="' . esc_attr( $mode ) . '">';
		$html .= '<header class="bqb-class__head">';
		$html .= '<h4>' . esc_html( $label ) . ' <span class="bqb-class__count" data-count>' . ( $assigned ? count( $assigned ) : '' ) . '</span></h4>';

		if ( 'bqb_theme' === $taxonomy ) {
			$html .= '<input type="search" class="bqb-class__filter" placeholder="Filtrer les thèmes…" data-filter />';
		}

		$html .= '</header>';

		if ( $help ) {
			$html .= '<p class="bqb-class__help">' . esc_html( $help ) . '</p>';
		}

		if ( 'tokens' === $mode ) {
			$html .= '<div class="bqb-tokens" data-tokens>';
			$html .= '<ul class="bqb-tokens__list" data-token-list>';

			foreach ( $assigned as $term_id ) {
				$term = get_term( $term_id, $taxonomy );
				if ( $term && ! is_wp_error( $term ) ) {
					$html .= '<li class="bqb-token"><input type="hidden" name="bqb_terms[' . esc_attr( $taxonomy ) . '][]" value="' . (int) $term->term_id . '" /><span>' . esc_html( $term->name ) . '</span><button type="button" aria-label="Retirer">×</button></li>';
				}
			}

			$html .= '</ul>';
			$html .= '<div class="bqb-tokens__search"><input type="text" class="bqb-admin__input" placeholder="Rechercher ou créer…" data-token-input autocomplete="off" /><div class="bqb-tokens__menu" data-token-menu hidden></div></div>';
			$html .= '</div>';

		} elseif ( 'tree' === $mode ) {
			$map   = bqb_term_children_map( $taxonomy );
			$html .= '<div class="bqb-tree-pick">';

			foreach ( isset( $map[0] ) ? $map[0] : array() as $root ) {
				$html .= '<div class="bqb-tree-pick__group" data-group>';
				$html .= '<div class="bqb-tree-pick__root">' . bqb_pill( $taxonomy, $root, in_array( (int) $root->term_id, $assigned, true ), 'bqb-pill--root' ) . '</div>';

				if ( ! empty( $map[ $root->term_id ] ) ) {
					$html .= '<div class="bqb-tree-pick__children">';
					foreach ( $map[ $root->term_id ] as $child ) {
						$html .= bqb_pill( $taxonomy, $child, in_array( (int) $child->term_id, $assigned, true ) );

						// Un troisième niveau reste possible, affiché à la suite.
						foreach ( isset( $map[ $child->term_id ] ) ? $map[ $child->term_id ] : array() as $grand ) {
							$html .= bqb_pill( $taxonomy, $grand, in_array( (int) $grand->term_id, $assigned, true ), 'bqb-pill--sub' );
						}
					}
					$html .= '</div>';
				}

				$html .= '</div>';
			}

			$html .= '</div>';

		} else {
			$map   = bqb_term_children_map( $taxonomy );
			$html .= '<div class="bqb-pills" data-pills>';

			foreach ( isset( $map[0] ) ? $map[0] : array() as $term ) {
				$html .= bqb_pill( $taxonomy, $term, in_array( (int) $term->term_id, $assigned, true ) );
			}

			$html .= '</div>';
		}

		if ( 'tokens' !== $mode ) {
			$html .= '<div class="bqb-class__add"><button type="button" class="bqb-linkbtn" data-add-open>+ Ajouter ' . esc_html( strtolower( $taxonomies[ $taxonomy ][1] ) ) . '</button>'
				. '<span class="bqb-class__addform" data-add-form hidden><input type="text" class="bqb-admin__input" placeholder="Nom" data-add-input /><button type="button" class="button" data-add-save>Créer</button></span></div>';
		}

		$html .= '</section>';
	}

	$html .= '<input type="hidden" name="bqb_terms_present" value="1" />';
	$html .= '<p class="bqb-class__foot">Pour réorganiser les catégories elles-mêmes (renommer, déplacer, supprimer), utilisez <a href="' . esc_url( admin_url( 'edit.php?post_type=' . BQB_CPT . '&page=bqb-arbo' ) ) . '">Bibliothèque › Arborescence</a>.</p>';
	$html .= '</div>';

	echo $html; // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
}

add_action( 'save_post_' . BQB_CPT, 'bqb_save_classement', 11, 2 );
function bqb_save_classement( $post_id, $post ) {
	if ( ! isset( $_POST['bqb_classement_nonce'] ) || ! wp_verify_nonce( sanitize_key( wp_unslash( $_POST['bqb_classement_nonce'] ) ), 'bqb_save_classement' ) ) {
		return;
	}

	if ( ( defined( 'DOING_AUTOSAVE' ) && DOING_AUTOSAVE ) || wp_is_post_revision( $post_id ) ) {
		return;
	}

	if ( ! current_user_can( 'edit_post', $post_id ) || empty( $_POST['bqb_terms_present'] ) ) {
		return;
	}

	$raw = isset( $_POST['bqb_terms'] ) && is_array( $_POST['bqb_terms'] ) ? wp_unslash( $_POST['bqb_terms'] ) : array(); // phpcs:ignore WordPress.Security.ValidatedSanitizedInput

	foreach ( array_keys( bqb_taxonomies() ) as $taxonomy ) {
		$ids = isset( $raw[ $taxonomy ] ) ? array_values( array_unique( array_filter( array_map( 'absint', (array) $raw[ $taxonomy ] ) ) ) ) : array();

		// Seuls des termes existants de la bonne famille sont acceptés.
		$ids = array_values(
			array_filter(
				$ids,
				function ( $id ) use ( $taxonomy ) {
					$term = get_term( $id, $taxonomy );
					return $term && ! is_wp_error( $term );
				}
			)
		);

		wp_set_object_terms( $post_id, $ids, $taxonomy, false );
	}
}

function bqb_ajax_guard( $taxonomy, $cap = 'assign' ) {
	check_ajax_referer( 'bqb_terms', 'nonce' );

	if ( ! array_key_exists( $taxonomy, bqb_taxonomies() ) ) {
		wp_send_json_error( array( 'message' => 'Famille inconnue.' ), 400 );
	}

	$tax  = get_taxonomy( $taxonomy );
	$need = ( 'manage' === $cap ) ? $tax->cap->manage_terms : $tax->cap->assign_terms;

	if ( ! current_user_can( $need ) ) {
		wp_send_json_error( array( 'message' => 'Droits insuffisants.' ), 403 );
	}
}

add_action( 'wp_ajax_bqb_term_search', 'bqb_ajax_term_search' );
function bqb_ajax_term_search() {
	$taxonomy = isset( $_GET['tax'] ) ? sanitize_key( wp_unslash( $_GET['tax'] ) ) : '';
	bqb_ajax_guard( $taxonomy );

	$q     = isset( $_GET['q'] ) ? sanitize_text_field( wp_unslash( $_GET['q'] ) ) : '';
	$terms = get_terms(
		array(
			'taxonomy'   => $taxonomy,
			'hide_empty' => false,
			'search'     => $q,
			'number'     => 15,
			'orderby'    => 'name',
		)
	);

	$out = array();
	foreach ( is_wp_error( $terms ) ? array() : $terms as $term ) {
		$out[] = array( 'id' => (int) $term->term_id, 'name' => html_entity_decode( $term->name, ENT_QUOTES, 'UTF-8' ) );
	}

	wp_send_json_success( $out );
}

add_action( 'wp_ajax_bqb_term_create', 'bqb_ajax_term_create' );
function bqb_ajax_term_create() {
	$taxonomy = isset( $_POST['tax'] ) ? sanitize_key( wp_unslash( $_POST['tax'] ) ) : '';
	bqb_ajax_guard( $taxonomy, 'manage' );

	$name   = isset( $_POST['name'] ) ? sanitize_text_field( wp_unslash( $_POST['name'] ) ) : '';
	$parent = isset( $_POST['parent'] ) ? absint( $_POST['parent'] ) : 0;

	if ( '' === $name ) {
		wp_send_json_error( array( 'message' => 'Le nom est vide.' ), 400 );
	}

	// Un terme du même nom existe déjà : on le réutilise, sans doublon.
	$existing = get_term_by( 'name', $name, $taxonomy );
	if ( $existing ) {
		wp_send_json_success( array( 'id' => (int) $existing->term_id, 'name' => $existing->name, 'existed' => true ) );
	}

	$created = wp_insert_term( $name, $taxonomy, array( 'parent' => $parent ) );

	if ( is_wp_error( $created ) ) {
		wp_send_json_error( array( 'message' => $created->get_error_message() ), 400 );
	}

	wp_send_json_success( array( 'id' => (int) $created['term_id'], 'name' => $name, 'existed' => false ) );
}

function bqb_render_arbo_node( $term, $taxonomy, $map ) {
	$tax      = get_taxonomy( $taxonomy );
	$short    = bqb_short_name( $taxonomy );
	$code     = '[bqp_documents ' . $short . '="' . $term->slug . '"]';
	$color    = ( 'bqb_collection' === $taxonomy && ! $term->parent ) ? get_term_meta( $term->term_id, 'bqb_couleur', true ) : '';
	$page     = (int) get_term_meta( $term->term_id, 'bqb_page', true );
	$children = isset( $map[ $term->term_id ] ) ? $map[ $term->term_id ] : array();
	$view     = $tax->public ? bqb_term_url( $term ) : '';

	$html  = '<li class="bqb-arbo__item" data-id="' . (int) $term->term_id . '">';
	$html .= '<div class="bqb-node" data-node>';
	$html .= '<span class="bqb-node__handle" title="Glisser pour déplacer" data-handle><span class="dashicons dashicons-menu"></span></span>';

	if ( $color ) {
		$html .= '<span class="bqb-node__swatch" style="background:' . esc_attr( $color ) . ';"></span>';
	}

	$html .= '<span class="bqb-node__name" data-name title="Cliquer pour renommer">' . esc_html( $term->name ) . '</span>';
	$html .= '<input type="text" class="bqb-node__rename" value="' . esc_attr( $term->name ) . '" data-rename hidden />';
	// Pour une arborescence, on compte aussi les documents des sous-catégories.
	$total = $tax->hierarchical ? bqb_count( array( $short => array( $term->slug ) ) ) : (int) $term->count;
	$html .= '<span class="bqb-node__count" title="Documents publiés, sous-catégories comprises">' . (int) $total . '</span>';

	if ( $page ) {
		$html .= '<span class="bqb-node__flag" title="Cette catégorie renvoie vers une page dédiée">Page dédiée</span>';
	}

	$html .= '<button type="button" class="bqb-node__code" data-copy="' . esc_attr( $code ) . '" title="Cliquer pour copier">' . esc_html( $code ) . '</button>';
	$html .= '<span class="bqb-node__actions">';
	$html .= '<button type="button" class="bqb-icon" data-act="rename" title="Renommer"><span class="dashicons dashicons-edit"></span></button>';

	if ( $tax->hierarchical ) {
		$html .= '<button type="button" class="bqb-icon" data-act="child" title="Ajouter une sous-catégorie"><span class="dashicons dashicons-plus-alt2"></span></button>';
	}

	$html .= '<button type="button" class="bqb-icon" data-act="panel" title="Réglages"><span class="dashicons dashicons-admin-generic"></span></button>';

	if ( $view ) {
		$html .= '<a class="bqb-icon" href="' . esc_url( $view ) . '" target="_blank" rel="noopener" title="Voir sur le site"><span class="dashicons dashicons-external"></span></a>';
	}

	$html .= '<button type="button" class="bqb-icon bqb-icon--danger" data-act="delete" data-count="' . (int) $term->count . '" data-children="' . count( $children ) . '" title="Supprimer"><span class="dashicons dashicons-trash"></span></button>';
	$html .= '</span></div>';
	$html .= '<div class="bqb-node__panel" data-panel hidden></div>';
	$html .= '<ul class="bqb-arbo__list" data-parent="' . (int) $term->term_id . '">';

	foreach ( $children as $child ) {
		$html .= bqb_render_arbo_node( $child, $taxonomy, $map );
	}

	$html .= '</ul></li>';

	return $html;
}

/* -------------------------------------------------------------------------
 * 19. Arborescence : les actions
 * ---------------------------------------------------------------------- */

add_action( 'wp_ajax_bqb_arbo', 'bqb_ajax_arbo' );
function bqb_ajax_arbo() {
	$taxonomy = isset( $_POST['tax'] ) ? sanitize_key( wp_unslash( $_POST['tax'] ) ) : '';
	bqb_ajax_guard( $taxonomy, 'manage' );

	$op = isset( $_POST['op'] ) ? sanitize_key( wp_unslash( $_POST['op'] ) ) : '';
	$id = isset( $_POST['id'] ) ? absint( $_POST['id'] ) : 0;

	$term = $id ? get_term( $id, $taxonomy ) : null;
	if ( $id && ( ! $term || is_wp_error( $term ) ) ) {
		wp_send_json_error( array( 'message' => 'Catégorie introuvable.' ), 404 );
	}

	switch ( $op ) {

		case 'add':
			$name   = isset( $_POST['name'] ) ? sanitize_text_field( wp_unslash( $_POST['name'] ) ) : '';
			$parent = isset( $_POST['parent'] ) ? absint( $_POST['parent'] ) : 0;

			if ( '' === $name ) {
				wp_send_json_error( array( 'message' => 'Le nom est vide.' ), 400 );
			}

			$created = wp_insert_term( $name, $taxonomy, array( 'parent' => $parent ) );
			if ( is_wp_error( $created ) ) {
				wp_send_json_error( array( 'message' => $created->get_error_message() ), 400 );
			}

			// Nouvelle catégorie rangée en dernier parmi ses sœurs.
			$siblings = get_terms( array( 'taxonomy' => $taxonomy, 'parent' => $parent, 'hide_empty' => false, 'fields' => 'ids' ) );
			update_term_meta( $created['term_id'], 'bqb_ordre', is_wp_error( $siblings ) ? 1 : count( $siblings ) );

			$new = get_term( $created['term_id'], $taxonomy );
			wp_send_json_success( array( 'html' => bqb_render_arbo_node( $new, $taxonomy, array() ) ) );

		case 'rename':
			$name = isset( $_POST['name'] ) ? sanitize_text_field( wp_unslash( $_POST['name'] ) ) : '';
			if ( '' === $name ) {
				wp_send_json_error( array( 'message' => 'Le nom est vide.' ), 400 );
			}

			// Seul le nom change : l'identifiant, donc les shortcodes, restent.
			$done = wp_update_term( $id, $taxonomy, array( 'name' => $name, 'slug' => $term->slug ) );
			if ( is_wp_error( $done ) ) {
				wp_send_json_error( array( 'message' => $done->get_error_message() ), 400 );
			}

			wp_send_json_success( array( 'name' => $name ) );

		case 'delete':
			$done = wp_delete_term( $id, $taxonomy );
			if ( ! $done || is_wp_error( $done ) ) {
				wp_send_json_error( array( 'message' => 'Suppression impossible.' ), 400 );
			}

			wp_send_json_success();

		case 'reorder':
			$ids = isset( $_POST['ids'] ) ? array_map( 'absint', (array) wp_unslash( $_POST['ids'] ) ) : array();
			$pos = 1;

			foreach ( $ids as $term_id ) {
				$t = get_term( $term_id, $taxonomy );
				if ( $t && ! is_wp_error( $t ) ) {
					update_term_meta( $term_id, 'bqb_ordre', $pos );
					$pos++;
				}
			}

			bqb_bump_cache();
			wp_send_json_success();

		case 'panel':
			wp_send_json_success( array( 'html' => bqb_render_arbo_panel( $term ) ) );

		case 'save':
			$data = array(
				'description' => isset( $_POST['description'] ) ? sanitize_textarea_field( wp_unslash( $_POST['description'] ) ) : $term->description,
				'slug'        => isset( $_POST['slug'] ) && '' !== trim( wp_unslash( $_POST['slug'] ) ) ? sanitize_title( wp_unslash( $_POST['slug'] ) ) : $term->slug,
			);

			if ( isset( $_POST['parent'] ) && get_taxonomy( $taxonomy )->hierarchical ) {
				$parent = absint( $_POST['parent'] );

				// Une catégorie ne peut pas être rangée sous elle-même ou sous une de ses descendantes.
				$descendants = get_term_children( $id, $taxonomy );
				if ( $parent !== $id && ! in_array( $parent, is_wp_error( $descendants ) ? array() : array_map( 'intval', $descendants ), true ) ) {
					$data['parent'] = $parent;
				}
			}

			$done = wp_update_term( $id, $taxonomy, $data );
			if ( is_wp_error( $done ) ) {
				wp_send_json_error( array( 'message' => $done->get_error_message() ), 400 );
			}

			if ( isset( $_POST['page'] ) ) {
				$page = absint( $_POST['page'] );
				$page ? update_term_meta( $id, 'bqb_page', $page ) : delete_term_meta( $id, 'bqb_page' );
			}

			if ( isset( $_POST['couleur'] ) ) {
				$color = sanitize_hex_color( wp_unslash( $_POST['couleur'] ) );
				if ( $color ) {
					update_term_meta( $id, 'bqb_couleur', $color );
				}
			}

			if ( 'bqb_collection' === $taxonomy ) {
				$natures = isset( $_POST['natures'] ) ? array_values( array_filter( array_map( 'absint', (array) wp_unslash( $_POST['natures'] ) ) ) ) : array();
				update_term_meta( $id, 'bqb_natures', $natures );

				$vues = isset( $_POST['vues'] ) ? array_values( array_intersect( array_map( 'sanitize_key', (array) wp_unslash( $_POST['vues'] ) ), array_keys( bqb_views() ) ) ) : array();
				update_term_meta( $id, 'bqb_vues', $vues );

				foreach ( array_keys( bqb_views() ) as $view ) {
					if ( isset( $_POST[ 'vue_' . $view ] ) ) {
						$label = sanitize_text_field( wp_unslash( $_POST[ 'vue_' . $view ] ) );
						$label ? update_term_meta( $id, 'bqb_vue_' . $view, $label ) : delete_term_meta( $id, 'bqb_vue_' . $view );
					}
				}
			}

			bqb_bump_cache();

			$moved = isset( $data['parent'] ) && (int) $data['parent'] !== (int) $term->parent;
			$fresh = get_term( $id, $taxonomy );

			wp_send_json_success(
				array(
					'reload' => $moved || $fresh->slug !== $term->slug,
					'name'   => $fresh->name,
				)
			);
	}

	wp_send_json_error( array( 'message' => 'Action inconnue.' ), 400 );
}

/**
 * Liste déroulante d'une famille, en arborescence, valeur = identifiant.
 */
function bqb_term_select( $taxonomy, $name, $selected = '', $all = 'Toutes' ) {
	$map  = bqb_term_children_map( $taxonomy );
	$html = '<select class="bqb-admin__input" name="' . esc_attr( $name ) . '" data-gen="' . esc_attr( bqb_short_name( $taxonomy ) ) . '"><option value="">' . esc_html( $all ) . '</option>';

	$walk = function ( $parent, $depth ) use ( &$walk, $map, $selected, &$html ) {
		foreach ( isset( $map[ $parent ] ) ? $map[ $parent ] : array() as $term ) {
			$html .= '<option value="' . esc_attr( $term->slug ) . '"' . selected( $selected, $term->slug, false ) . '>' . str_repeat( '&nbsp;&nbsp;&nbsp;', $depth ) . esc_html( $term->name ) . '</option>';
			$walk( $term->term_id, $depth + 1 );
		}
	};
	$walk( 0, 0 );

	return $html . '</select>';
}

		if ( ! empty( $_GET[ $short ] ) ) {

}

function bqb_access_badge( $post_id ) {
	$acces  = get_post_meta( $post_id, '_bqb_acces', true );
	$acces  = $acces ? $acces : 'public';
	$labels = array(
		'public'      => 'Accès libre',
		'adherents'   => 'Adhérents',
		'partenaires' => 'Partenaires',
		'sur-place'   => 'Sur place',
	);

	return '<span class="bqb-badge bqb-badge--' . esc_attr( $acces ) . '">' . esc_html( isset( $labels[ $acces ] ) ? $labels[ $acces ] : 'Accès libre' ) . '</span>';
}

function bqb_is_public( $post_id ) {
	$acces = get_post_meta( $post_id, '_bqb_acces', true );

	return ! $acces || 'public' === $acces;
}

function bqb_doc_date( $post_id ) {
	$annee = get_post_meta( $post_id, '_bqb_annee', true );
	$mois  = (int) get_post_meta( $post_id, '_bqb_mois', true );
	$jour  = (int) get_post_meta( $post_id, '_bqb_jour', true );

	if ( ! $annee ) {
		return '';
	}

	$months = bqb_months();

	if ( $mois && $jour ) {
		return $jour . ' ' . $months[ $mois ] . ' ' . $annee;
	}

	return $mois ? $months[ $mois ] . ' ' . $annee : (string) $annee;
}

function bqb_term_names( $post_id, $taxonomy, $linked = false, $limit = 0 ) {
	$terms = get_the_terms( $post_id, $taxonomy );

	if ( ! $terms || is_wp_error( $terms ) ) {
		return array();
	}

	$out = array();
	foreach ( $terms as $term ) {
		$url   = $linked ? bqb_term_url( $term ) : '';
		$out[] = $url ? '<a href="' . esc_url( $url ) . '">' . esc_html( $term->name ) . '</a>' : esc_html( $term->name );

		if ( $limit && count( $out ) >= $limit ) {
			break;
		}
	}

	return $out;
}

function bqb_is_video( $url ) {
	return (bool) preg_match( '#(youtube\.com|youtu\.be|vimeo\.com|dailymotion\.com)#i', (string) $url );
}

/**
 * Le bouton d'action d'un document : télécharger, regarder, consulter,
 * ou voir la fiche quand l'accès est réservé.
 */
function bqb_doc_action( $post_id, $context = 'list' ) {
	$file = (int) get_post_meta( $post_id, '_bqb_fichier', true );
	$link = get_post_meta( $post_id, '_bqb_lien', true );

	if ( ! bqb_is_public( $post_id ) ) {
		if ( 'list' === $context ) {
			return array( get_permalink( $post_id ), 'Voir la fiche', false );
		}

		$contact = (int) bqb_settings()['page_contact'];

		return $contact ? array( add_query_arg( 'document', rawurlencode( get_the_title( $post_id ) ), get_permalink( $contact ) ), 'Demander l\'accès', false ) : array( '', '', false );
	}

	if ( $file && wp_get_attachment_url( $file ) ) {
		return array( wp_get_attachment_url( $file ), 'Télécharger', true );
	}

	if ( $link ) {
		return array( $link, bqb_is_video( $link ) ? 'Regarder' : 'Consulter la source', false );
	}

	return 'list' === $context ? array( get_permalink( $post_id ), 'Voir la fiche', false ) : array( '', '', false );
}

function bqb_render_doc_row( $post_id ) {
	$collections = bqb_term_names( $post_id, 'bqb_collection', false, 1 );
	$authors     = bqb_term_names( $post_id, 'bqb_personne', false, 3 );
	$natures     = bqb_term_names( $post_id, 'bqb_nature', false, 1 );
	$themes      = bqb_term_names( $post_id, 'bqb_theme', false, 1 );
	$date        = bqb_doc_date( $post_id );
	$file        = (int) get_post_meta( $post_id, '_bqb_fichier', true );

	$info = array_filter(
		array(
			implode( ', ', $authors ),
			$date ? esc_html( $date ) : '',
			$natures ? $natures[0] : '',
			$themes ? $themes[0] : '',
			( $file && bqb_is_public( $post_id ) ) ? esc_html( str_replace( ' · ', ', ', bqb_file_label( $file ) ) ) : '',
		)
	);

	list( $url, $label, $download ) = bqb_doc_action( $post_id );

	$html  = '<article class="bqb-doc">';

	if ( has_post_thumbnail( $post_id ) ) {
		$html .= '<a class="bqb-doc__cover" href="' . esc_url( get_permalink( $post_id ) ) . '" tabindex="-1" aria-hidden="true">' . get_the_post_thumbnail( $post_id, 'thumbnail', array( 'alt' => '' ) ) . '</a>';
	}

	$html .= '<div class="bqb-doc__main">';
	$html .= '<div class="bqb-doc__meta">' . bqb_access_badge( $post_id ) . ( $collections ? '<span class="bqb-doc__coll">' . $collections[0] . '</span>' : '' ) . '</div>';
	$html .= '<h3 class="bqb-doc__title"><a href="' . esc_url( get_permalink( $post_id ) ) . '">' . esc_html( get_the_title( $post_id ) ) . '</a></h3>';

	if ( $info ) {
		$html .= '<p class="bqb-doc__info">' . implode( '<span class="bqb-sep">·</span>', $info ) . '</p>';
	}

	$html .= '</div>';

	if ( $url ) {
		$html .= '<div class="bqb-doc__action"><a class="bqb-btn bqb-btn--ghost" href="' . esc_url( $url ) . '"' . ( $download ? ' download' : '' ) . ( ( ! $download && $url !== get_permalink( $post_id ) ) ? ' target="_blank" rel="noopener"' : '' ) . '>' . esc_html( $label ) . '</a></div>';
	}

	$html .= '</article>';

	return $html;
}

			'filtres'    => 'non',
			'titre'      => '',
			'vide'       => 'Aucun document ne correspond pour le moment.',
			'pagination' => 'oui',
		)
	);

	return bqb_render_documents( shortcode_atts( $defaults, $atts, 'bqp_documents' ) );
}

/**
 * Les années présentes dans la bibliothèque, pour le filtre.
 */
function bqb_years() {
	$key    = 'bqb_years_' . md5( (string) get_option( 'bqb_cache_version', '1' ) );
	$cached = get_transient( $key );

	if ( false !== $cached ) {
		return $cached;
	}

	global $wpdb;
	$years = $wpdb->get_col( $wpdb->prepare( "SELECT DISTINCT pm.meta_value FROM {$wpdb->postmeta} pm INNER JOIN {$wpdb->posts} p ON p.ID = pm.post_id WHERE pm.meta_key = '_bqb_annee' AND p.post_type = %s AND p.post_status = 'publish' ORDER BY pm.meta_value DESC", BQB_CPT ) ); // phpcs:ignore WordPress.DB.DirectDatabaseQuery

	$years = array_values( array_filter( array_map( 'intval', $years ) ) );
	rsort( $years );

	set_transient( $key, $years, DAY_IN_SECONDS );

	return $years;
}

function bqb_url_filters() {
	$out = array();

	foreach ( array_merge( array_keys( bqb_short_map() ), array( 'q', 'annee', 'tri', 'groupe' ) ) as $key ) {
		if ( isset( $_GET[ 'f_' . $key ] ) && '' !== $_GET[ 'f_' . $key ] ) { // phpcs:ignore WordPress.Security.NonceVerification
			$out[ $key ] = sanitize_text_field( wp_unslash( $_GET[ 'f_' . $key ] ) ); // phpcs:ignore WordPress.Security.NonceVerification
		}
	}

	return $out;
}

function bqb_render_documents( $atts, $locked = array() ) {
	static $instance = 0;
	$instance++;

	// Seule la première liste de la page écoute l'adresse (filtres, page).
	$listens = ( 1 === $instance );
	$url_f   = $listens ? bqb_url_filters() : array();
	$groups  = array( 'annee', 'personne', 'theme', 'organisation', 'nature', 'collection' );

	$criteria = array();
	foreach ( array_keys( bqb_short_map() ) as $short ) {
		$value = isset( $locked[ $short ] ) ? $locked[ $short ] : $atts[ $short ];
		if ( '' !== $value && array() !== $value ) {
			$criteria[ $short ] = is_array( $value ) ? $value : array_map( 'trim', explode( ',', $value ) );
		}
	}

	$criteria['and'] = array();
	foreach ( array_keys( bqb_short_map() ) as $short ) {
		if ( isset( $url_f[ $short ] ) ) {
			$criteria['and'][ $short ] = array( $url_f[ $short ] );
		}
	}

	$criteria['q']     = isset( $url_f['q'] ) ? $url_f['q'] : '';
	$criteria['annee'] = isset( $url_f['annee'] ) ? $url_f['annee'] : $atts['annee'];
	$criteria['acces'] = $atts['acces'];
	$criteria['tri']   = isset( $url_f['tri'] ) ? $url_f['tri'] : ( $atts['tri'] ? $atts['tri'] : 'recent' );

	$groupe = isset( $url_f['groupe'] ) ? $url_f['groupe'] : $atts['groupe'];
	$groupe = in_array( $groupe, $groups, true ) ? $groupe : '';

	$settings = bqb_settings();
	$per_page = $atts['nombre'] ? max( 1, (int) $atts['nombre'] ) : ( $groupe ? 300 : (int) $settings['par_page'] );
	$paged    = ( $listens && isset( $_GET['pg'] ) ) ? max( 1, absint( $_GET['pg'] ) ) : 1; // phpcs:ignore WordPress.Security.NonceVerification

	$args                   = bqb_query_args( $criteria );
	$args['posts_per_page'] = $per_page;
	$args['paged']          = $paged;

	$query = new WP_Query( $args );

	wp_enqueue_style( 'bqb-front' );

	$html = '<section class="bqb-docs">';

	if ( '' !== trim( $atts['titre'] ) ) {
		$html .= '<h2 class="bqb-docs__title">' . esc_html( $atts['titre'] ) . '</h2>';
	}

	if ( 'oui' === strtolower( $atts['filtres'] ) ) {
		$html .= bqb_render_filter_bar( $criteria, $url_f, $locked + array_filter( array_intersect_key( $atts, bqb_short_map() ) ) );
	} elseif ( $url_f ) {
		$html .= bqb_render_active_filters( $url_f );
	}

	$total = (int) $query->found_posts;

	if ( $total ) {
		$html .= '<p class="bqb-docs__count">' . sprintf( _n( '%s document', '%s documents', $total ), number_format_i18n( $total ) );

		if ( $groupe ) {
			$names  = array( 'annee' => 'par année', 'personne' => 'par auteur', 'theme' => 'par thème', 'organisation' => 'par organisation', 'nature' => 'par type', 'collection' => 'par collection' );
			$html  .= ', regroupés ' . esc_html( $names[ $groupe ] );
		}

		$html .= '</p>';
	}

	if ( ! $query->have_posts() ) {
		$html .= '<p class="bqb-docs__empty">' . esc_html( $atts['vide'] ) . '</p></section>';
		wp_reset_postdata();
		return $html;
	}

	$ids = wp_list_pluck( $query->posts, 'ID' );

	if ( $groupe ) {
		$html .= bqb_render_grouped( $ids, $groupe, $criteria['tri'] );
	} else {
		$html .= '<div class="bqb-docs__list">';
		foreach ( $ids as $id ) {
			$html .= bqb_render_doc_row( $id );
		}
		$html .= '</div>';
	}

	if ( 'oui' === strtolower( $atts['pagination'] ) && $query->max_num_pages > 1 && $listens ) {
		$html .= '<nav class="bqb-pages" aria-label="Pages de résultats">';
		for ( $i = 1; $i <= $query->max_num_pages; $i++ ) {
			$html .= ( $i === $paged )
				? '<span class="bqb-pages__item is-on" aria-current="page">' . $i . '</span>'
				: '<a class="bqb-pages__item" href="' . esc_url( add_query_arg( 'pg', $i ) ) . '">' . $i . '</a>';
		}
		$html .= '</nav>';
	}

	$html .= '</section>';
	wp_reset_postdata();

	return $html;
}

function bqb_render_grouped( $ids, $groupe, $tri ) {
	$buckets = array();
	$other   = 'annee' === $groupe ? 'Sans date' : 'Non renseigné';

	foreach ( $ids as $id ) {
		if ( 'annee' === $groupe ) {
			$year                  = get_post_meta( $id, '_bqb_annee', true );
			$buckets[ $year ? (string) $year : $other ][] = $id;
			continue;
		}

		$map   = bqb_short_map();
		$terms = get_the_terms( $id, $map[ $groupe ] );

		if ( ! $terms || is_wp_error( $terms ) ) {
			$buckets[ $other ][] = $id;
			continue;
		}

		foreach ( $terms as $term ) {
			$buckets[ $term->name ][] = $id;
		}
	}

	$keys = array_keys( $buckets );

	if ( 'annee' === $groupe ) {
		usort(
			$keys,
			function ( $a, $b ) use ( $other, $tri ) {
				if ( $a === $other ) { return 1; }
				if ( $b === $other ) { return -1; }
				return 'ancien' === $tri ? ( (int) $a - (int) $b ) : ( (int) $b - (int) $a );
			}
		);
	} else {
		usort(
			$keys,
			function ( $a, $b ) use ( $other ) {
				if ( $a === $other ) { return 1; }
				if ( $b === $other ) { return -1; }
				return strcasecmp( remove_accents( $a ), remove_accents( $b ) );
			}
		);
	}

	$html = '<div class="bqb-groups">';
	foreach ( $keys as $key ) {
		$html .= '<div class="bqb-group"><h3 class="bqb-group__title">' . esc_html( $key ) . ' <span>' . count( $buckets[ $key ] ) . '</span></h3><div class="bqb-docs__list">';
		foreach ( $buckets[ $key ] as $id ) {
			$html .= bqb_render_doc_row( $id );
		}
		$html .= '</div></div>';
	}

	return $html . '</div>';
}

/**
 * Les filtres venus d'un lien (par exemple depuis un bloc), affichés en
 * étiquettes que l'on peut retirer une à une.
 */
function bqb_render_active_filters( $url_f ) {
	$map   = bqb_short_map();
	$chips = array();

	foreach ( $url_f as $key => $value ) {
		if ( isset( $map[ $key ] ) ) {
			$term  = get_term_by( 'slug', $value, $map[ $key ] );
			$label = $term ? $term->name : $value;
		} elseif ( 'groupe' === $key ) {
			$names = array( 'annee' => 'Par année', 'personne' => 'Par auteur', 'theme' => 'Par thème', 'organisation' => 'Par organisation', 'nature' => 'Par type', 'collection' => 'Par collection' );
			$label = isset( $names[ $value ] ) ? $names[ $value ] : $value;
		} elseif ( 'q' === $key ) {
			$label = '« ' . $value . ' »';
		} elseif ( 'tri' === $key ) {
			continue;
		} else {
			$label = $value;
		}

		$chips[] = '<a class="bqb-chip" href="' . esc_url( remove_query_arg( array( 'f_' . $key, 'pg' ) ) ) . '">' . esc_html( $label ) . ' <span aria-hidden="true">×</span><span class="screen-reader-text"> (retirer ce filtre)</span></a>';
	}

	return $chips ? '<div class="bqb-active">' . implode( '', $chips ) . '</div>' : '';
}

function bqb_render_filter_bar( $criteria, $url_f, $locked ) {
	$action = remove_query_arg( array_merge( array_map( function ( $k ) { return 'f_' . $k; }, array_merge( array_keys( bqb_short_map() ), array( 'q', 'annee', 'tri', 'groupe' ) ) ), array( 'pg' ) ) );
	$val    = function ( $key ) use ( $url_f ) {
		return isset( $url_f[ $key ] ) ? $url_f[ $key ] : '';
	};

	$select = function ( $short, $label ) use ( $val ) {
		$map = bqb_short_map();
		return '<label class="bqb-filter"><span>' . esc_html( $label ) . '</span>' . str_replace( 'bqb-admin__input', 'bqb-input', bqb_term_select( $map[ $short ], 'f_' . $short, $val( $short ), 'Tous' ) ) . '</label>';
	};

	$html  = '<form class="bqb-filters" method="get" action="' . esc_url( $action ) . '">';
	$html .= '<div class="bqb-filters__row">';
	$html .= '<label class="bqb-filter bqb-filter--q"><span>Recherche libre</span><input type="search" class="bqb-input" name="f_q" value="' . esc_attr( $val( 'q' ) ) . '" placeholder="Titre, auteur, mot-clé…" /></label>';

	if ( empty( $locked['collection'] ) ) {
		$html .= $select( 'collection', 'Collection' );
	}

	$html .= $select( 'nature', 'Type de document' );
	$html .= $select( 'theme', 'Thème' );

	$html .= '<label class="bqb-filter"><span>Année</span><select class="bqb-input" name="f_annee"><option value="">Toutes</option>';
	foreach ( bqb_years() as $year ) {
		$html .= '<option value="' . (int) $year . '"' . selected( $val( 'annee' ), (string) $year, false ) . '>' . (int) $year . '</option>';
	}
	$html .= '</select></label>';

	$html .= '</div>';

	$more_open = $val( 'secteur' ) || $val( 'pays' ) || $val( 'prix' ) || $val( 'organisation' );
	$html     .= '<details class="bqb-filters__more"' . ( $more_open ? ' open' : '' ) . '><summary>Plus de filtres</summary><div class="bqb-filters__row">';
	$html     .= $select( 'secteur', 'Secteur' );
	$html     .= $select( 'pays', 'Pays ou territoire' );
	$html     .= $select( 'prix', 'Prix ou bourse' );
	$html     .= $select( 'organisation', 'Organisation' );
	$html     .= '</div></details>';

	$html .= '<div class="bqb-filters__foot">';
	$html .= '<label class="bqb-filter bqb-filter--inline"><span>Trier</span><select class="bqb-input" name="f_tri"><option value="">Plus récents</option><option value="ancien"' . selected( $val( 'tri' ), 'ancien', false ) . '>Plus anciens</option><option value="titre"' . selected( $val( 'tri' ), 'titre', false ) . '>A à Z</option></select></label>';

	if ( $val( 'groupe' ) ) {
		$html .= '<input type="hidden" name="f_groupe" value="' . esc_attr( $val( 'groupe' ) ) . '" />';
	}

	$html .= '<span class="bqb-filters__btns"><a class="bqb-btn bqb-btn--link" href="' . esc_url( $action ) . '">Réinitialiser</a><button type="submit" class="bqb-btn">Rechercher</button></span>';
	$html .= '</div></form>';

	return $html;
}

/* -------------------------------------------------------------------------
 * 24. [bqp_bibliotheque] : les cinq blocs
 * ---------------------------------------------------------------------- */

function bqb_laureats_count() {
	$terms = get_terms(
		array(
			'taxonomy'   => 'bqb_personne',
			'hide_empty' => false,
			'fields'     => 'ids',
			'meta_query' => array( array( 'key' => 'bqb_roles', 'value' => '"laureat"', 'compare' => 'LIKE' ) ),
		)
	);

	return is_wp_error( $terms ) ? 0 : count( $terms );
}

/**
 * Un lien de bloc, masqué s'il ne mène à rien et que le réglage le demande.
 */
function bqb_block_link( $url, $label, $count, $show_empty, $class = '' ) {
	if ( ! $url || ( ! $count && ! $show_empty ) ) {
		return '';
	}

	return '<li class="bqb-blink' . ( $class ? ' ' . $class : '' ) . ( $count ? '' : ' is-empty' ) . '"><a href="' . esc_url( $url ) . '">' . esc_html( $label ) . '</a>' . ( $count ? '<span class="bqb-blink__n">' . (int) $count . '</span>' : '' ) . '</li>';
}

function bqb_collection_items( $term, $show_empty, $split = false ) {
	$items    = '';
	$types    = '';
	$settings = bqb_settings();
	$slug     = $term->slug;

	foreach ( (array) get_term_meta( $term->term_id, 'bqb_vues', true ) as $view ) {
		if ( 'laureats' === $view ) {
			$page = (int) $settings['page_personnes'];
			if ( $page ) {
				$items .= bqb_block_link( add_query_arg( 'role', 'laureat', get_permalink( $page ) ), bqb_view_label( $view, $term ), bqb_laureats_count(), $show_empty, 'is-view' );
			}
			continue;
		}

		if ( array_key_exists( $view, bqb_views() ) ) {
			$items .= bqb_block_link( bqb_term_url( $term, array( 'f_groupe' => $view ) ), bqb_view_label( $view, $term ), bqb_count( array( 'collection' => array( $slug ) ) ), $show_empty, 'is-view' );
		}
	}

	foreach ( (array) get_term_meta( $term->term_id, 'bqb_natures', true ) as $nature_id ) {
		$nature = get_term( (int) $nature_id, 'bqb_nature' );

		if ( ! $nature || is_wp_error( $nature ) ) {
			continue;
		}

		$count  = bqb_count( array( 'collection' => array( $slug ), 'nature' => array( $nature->slug ) ) );
		$types .= bqb_block_link( bqb_term_url( $term, array( 'f_nature' => $nature->slug ) ), $nature->name, $count, $show_empty );
	}

	// Les vues (regroupements) et les types de documents peuvent être
	// rendus séparément, pour ne titrer « Types de documents » que les types.
	return $split ? array( $items, $types ) : $items . $types;
}

function bqb_sc_blocks( $atts ) {
	$atts = shortcode_atts( array( 'vides' => '', 'navigation' => 'oui' ), $atts, 'bqp_bibliotheque' );

	$settings   = bqb_settings();
	$show_empty = ( '' !== $atts['vides'] ) ? in_array( strtolower( $atts['vides'] ), array( 'oui', 'afficher', '1' ), true ) : (bool) $settings['afficher_vides'];
	$map        = bqb_term_children_map( 'bqb_collection' );

	wp_enqueue_style( 'bqb-front' );

	$html = '<div class="bqb-lib">';

	if ( 'non' !== strtolower( $atts['navigation'] ) ) {
		$html .= bqb_render_lib_nav();
	}

	$html .= '<div class="bqb-blocks">';
	$num   = 0;

	foreach ( isset( $map[0] ) ? $map[0] : array() as $root ) {
		$num++;
		$color = get_term_meta( $root->term_id, 'bqb_couleur', true );
		$color = $color ? $color : bqb_palette()['bordeaux'];
		$total = bqb_count( array( 'collection' => array( $root->slug ) ) );

		$body = '';

		// Liens portés par l'espace lui-même (ex. « Types de publications »).
		list( $own_views, $own_types ) = bqb_collection_items( $root, $show_empty, true );
		if ( $own_views ) {
			$body .= '<div class="bqb-sub"><ul class="bqb-sub__links">' . $own_views . '</ul></div>';
		}
		if ( $own_types ) {
			$body .= '<div class="bqb-sub"><span class="bqb-sub__label">Types de documents</span><ul class="bqb-sub__links">' . $own_types . '</ul></div>';
		}

		$sub = 0;
		foreach ( isset( $map[ $root->term_id ] ) ? $map[ $root->term_id ] : array() as $child ) {
			$count = bqb_count( array( 'collection' => array( $child->slug ) ) );

			if ( ! $count && ! $show_empty ) {
				continue;
			}

			$sub++;
			$items = bqb_collection_items( $child, $show_empty );
			$body .= '<div class="bqb-sub">';
			$body .= '<h4 class="bqb-sub__title"><a href="' . esc_url( bqb_term_url( $child ) ) . '"><span class="bqb-sub__num">' . $num . '.' . $sub . '</span>' . esc_html( $child->name ) . '</a>' . ( $count ? '<span class="bqb-blink__n">' . $count . '</span>' : '' ) . '</h4>';

			if ( $items ) {
				$body .= '<ul class="bqb-sub__links">' . $items . '</ul>';
			}

			$body .= '</div>';
		}

		$html .= bqb_render_block_card( $num, $root->name, $root->description, $color, ( $total || $show_empty ) ? bqb_term_url( $root ) : '', $body );
	}

	// Cinquième bloc : les thèmes.
	$num++;
	$themes = bqb_term_children_map( 'bqb_theme' );
	$body   = '';

	foreach ( isset( $themes[0] ) ? $themes[0] : array() as $root ) {
		$count = bqb_count( array( 'theme' => array( $root->slug ) ) );

		if ( ! $count && ! $show_empty ) {
			continue;
		}

		$items = '';
		foreach ( isset( $themes[ $root->term_id ] ) ? $themes[ $root->term_id ] : array() as $child ) {
			$items .= bqb_block_link( bqb_term_url( $child ), $child->name, bqb_count( array( 'theme' => array( $child->slug ) ) ), $show_empty );
		}

		$body .= '<div class="bqb-sub"><h4 class="bqb-sub__title"><a href="' . esc_url( bqb_term_url( $root ) ) . '">' . esc_html( $root->name ) . '</a>' . ( $count ? '<span class="bqb-blink__n">' . $count . '</span>' : '' ) . '</h4>' . ( $items ? '<ul class="bqb-sub__links is-inline">' . $items . '</ul>' : '' ) . '</div>';
	}

	if ( $settings['page_catalogue'] ) {
		$catalogue = get_permalink( (int) $settings['page_catalogue'] );
		$body     .= '<div class="bqb-sub"><h4 class="bqb-sub__title">Autres filtres</h4><ul class="bqb-sub__links">'
			. '<li class="bqb-blink is-view"><a href="' . esc_url( $catalogue ) . '">Secteurs, pays, périodes, types, collections</a></li>'
			. '</ul></div>';
	}

	$html .= bqb_render_block_card( $num, $settings['bloc5_titre'], $settings['bloc5_texte'], $settings['bloc5_couleur'], '', $body, 'bqb-block--wide' );
	$html .= '</div></div>';

	return $html;
}

function bqb_render_block_card( $num, $title, $text, $color, $url, $body, $class = '' ) {
	$html  = '<article class="bqb-block' . ( $class ? ' ' . esc_attr( $class ) : '' ) . '" style="--bqb-block:' . esc_attr( $color ) . ';--bqb-block-soft:' . esc_attr( bqb_hex_to_rgba( $color, 0.06 ) ) . ';--bqb-block-line:' . esc_attr( bqb_hex_to_rgba( $color, 0.18 ) ) . ';">';
	$html .= '<header class="bqb-block__head">';
	$html .= '<span class="bqb-block__num">' . (int) $num . '</span>';
	$html .= '<h3 class="bqb-block__title">' . ( $url ? '<a href="' . esc_url( $url ) . '">' . esc_html( $title ) . '</a>' : esc_html( $title ) ) . '</h3>';

	if ( $text ) {
		$html .= '<p class="bqb-block__text">' . esc_html( $text ) . '</p>';
	}

	$html .= '</header>';
	$html .= '<div class="bqb-block__body">' . ( $body ? $body : '<p class="bqb-block__soon">Les premiers documents arrivent bientôt.</p>' ) . '</div>';
	$html .= '</article>';

	return $html;
}

/**

	$s     = bqb_settings();

	if ( ! $links && ! $catalogue ) {
		return '';
	}

	$icons = array(

/* -------------------------------------------------------------------------
 * 25. Fil d'Ariane
 * ---------------------------------------------------------------------- */

function bqb_breadcrumb( $trail ) {
	$s    = bqb_settings();
	$home = $s['page_bibliotheque'] ? array( get_permalink( (int) $s['page_bibliotheque'] ), 'Bibliothèque' ) : null;

	if ( $home ) {
		array_unshift( $trail, $home );
	}

	if ( count( $trail ) < 2 ) {
		return '';
	}

	$items = array();
	$last  = count( $trail ) - 1;

	foreach ( $trail as $i => $crumb ) {
		$items[] = ( $i === $last || ! $crumb[0] )
			? '<span aria-current="page">' . esc_html( $crumb[1] ) . '</span>'
			: '<a href="' . esc_url( $crumb[0] ) . '">' . esc_html( $crumb[1] ) . '</a>';
	}

	return '<nav class="bqb-crumbs" aria-label="Fil d\'Ariane">' . implode( '<span class="bqb-crumbs__sep" aria-hidden="true">›</span>', $items ) . '</nav>';
}

function bqb_term_trail( $term ) {
	$trail = array();

	foreach ( array_reverse( get_ancestors( $term->term_id, $term->taxonomy, 'taxonomy' ) ) as $ancestor_id ) {
		$ancestor = get_term( $ancestor_id, $term->taxonomy );
		if ( $ancestor && ! is_wp_error( $ancestor ) ) {
			$trail[] = array( bqb_term_url( $ancestor ), $ancestor->name );
		}
	}

	$trail[] = array( bqb_term_url( $term ), $term->name );

	return $trail;
}

function bqb_related_ids( $post_id ) {
	$out = get_post_meta( $post_id, '_bqb_associes', true );
	$out = is_array( $out ) ? array_map( 'intval', $out ) : array();

	// Les documents qui citent celui-ci comme associé, en retour.
	$back = get_posts(
		array(
			'post_type'   => BQB_CPT,
			'post_status' => 'publish',
			'numberposts' => 20,
			'fields'      => 'ids',
			'meta_query'  => array( array( 'key' => '_bqb_associes', 'value' => 'i:' . (int) $post_id . ';', 'compare' => 'LIKE' ) ),
		)
	);

	foreach ( $back as $id ) {
		$list = get_post_meta( $id, '_bqb_associes', true );
		if ( is_array( $list ) && in_array( (int) $post_id, array_map( 'intval', $list ), true ) ) {
			$out[] = (int) $id;
		}
	}

	return array_values(
		array_unique(
			array_filter(
				$out,
				function ( $id ) use ( $post_id ) {
					return $id !== (int) $post_id && 'publish' === get_post_status( $id );
				}
			)
		)
	);
}

function bqb_render_fiche( $post_id, $notice_html ) {
	$excerpt  = get_post_field( 'post_excerpt', $post_id );
	$date     = bqb_doc_date( $post_id );
	$link     = get_post_meta( $post_id, '_bqb_lien', true );
	$file     = (int) get_post_meta( $post_id, '_bqb_fichier', true );
	$ref      = get_post_meta( $post_id, '_bqb_reference', true );
	$droits   = get_post_meta( $post_id, '_bqb_droits', true );
	$edition  = get_post_meta( $post_id, '_bqb_edition', true );
	$public   = bqb_is_public( $post_id );
	$colls    = get_the_terms( $post_id, 'bqb_collection' );
	$authors  = bqb_term_names( $post_id, 'bqb_personne', true );

	list( $url, $label, $download ) = bqb_doc_action( $post_id, 'fiche' );

	wp_enqueue_style( 'bqb-front' );
	wp_enqueue_script( 'bqb-front' );

	$html = '<div class="bqb-fiche">';

	if ( $colls && ! is_wp_error( $colls ) ) {
		$trail   = bqb_term_trail( $colls[0] );
		$trail[] = array( '', get_the_title( $post_id ) );
		$html   .= bqb_breadcrumb( $trail );
	}

	$html .= '<div class="bqb-fiche__top">';

	if ( has_post_thumbnail( $post_id ) ) {
		$html .= '<div class="bqb-fiche__cover">' . get_the_post_thumbnail( $post_id, 'medium', array( 'alt' => 'Couverture : ' . get_the_title( $post_id ) ) ) . '</div>';
	}

	$html .= '<div class="bqb-fiche__head">';
	$html .= '<div class="bqb-doc__meta">' . bqb_access_badge( $post_id ) . ( $colls && ! is_wp_error( $colls ) ? '<a class="bqb-doc__coll" href="' . esc_url( bqb_term_url( $colls[0] ) ) . '">' . esc_html( $colls[0]->name ) . '</a>' : '' ) . '</div>';

	if ( $authors || $date ) {
		$html .= '<p class="bqb-fiche__byline">' . implode( ', ', $authors ) . ( $authors && $date ? '<span class="bqb-sep">·</span>' : '' ) . esc_html( $date ) . '</p>';
	}

	if ( $excerpt ) {
		$html .= '<p class="bqb-fiche__lead">' . esc_html( $excerpt ) . '</p>';
	}

	$html .= '<div class="bqb-fiche__actions">';

	if ( $url ) {
		$extra = ( $download && $file ) ? ' <span class="bqb-btn__meta">' . esc_html( str_replace( ' · ', ', ', bqb_file_label( $file ) ) ) . '</span>' : '';
		$html .= '<a class="bqb-btn" href="' . esc_url( $url ) . '"' . ( $download ? ' download' : ' target="_blank" rel="noopener"' ) . '>' . esc_html( $label ) . $extra . '</a>';
	} elseif ( ! $public ) {
		$html .= '<p class="bqb-fiche__locked">Ce document est réservé. Contactez-nous pour en obtenir l\'accès.</p>';
	}

	$cite = trim( wp_strip_all_tags( implode( ', ', $authors ) ) . ( $authors ? ', ' : '' ) . '« ' . get_the_title( $post_id ) . ' »' . ( $ref ? ', ' . $ref : ( $date ? ', ' . $date : '' ) ) . '. ' . get_permalink( $post_id ) );
	$html .= '<button type="button" class="bqb-btn bqb-btn--ghost" data-cite="' . esc_attr( $cite ) . '">Citer ce document</button>';
	$html .= '</div></div></div>';

	if ( $public && $link && bqb_is_video( $link ) ) {
		$embed = wp_oembed_get( $link );
		if ( $embed ) {
			$html .= '<div class="bqb-fiche__video">' . $embed . '</div>';
		}
	}

	$rows = array(
		'Date'                => $date ? esc_html( $date ) : '',
		'Auteurs'             => implode( ', ', $authors ),
		'Type de document'    => implode( ', ', bqb_term_names( $post_id, 'bqb_nature', true ) ),
		'Collection'          => implode( ', ', bqb_term_names( $post_id, 'bqb_collection', true ) ),
		'Thèmes'              => implode( ', ', bqb_term_names( $post_id, 'bqb_theme', true ) ),
		'Secteurs'            => implode( ', ', bqb_term_names( $post_id, 'bqb_secteur', true ) ),
		'Pays et territoires' => implode( ', ', bqb_term_names( $post_id, 'bqb_pays', true ) ),
		'Organisations'       => implode( ', ', bqb_term_names( $post_id, 'bqb_organisation', true ) ),
		'Prix ou bourse'      => implode( ', ', bqb_term_names( $post_id, 'bqb_prix', true ) ) . ( $edition ? ', édition ' . (int) $edition : '' ),
		'Référence'           => $ref ? esc_html( $ref ) : '',
		'Droits'              => $droits ? esc_html( $droits ) : '',
	);

	$dl = '';
	foreach ( $rows as $label_row => $value ) {
		if ( '' !== trim( wp_strip_all_tags( $value ), ', ' ) ) {
			$dl .= '<div class="bqb-fiche__row"><dt>' . esc_html( $label_row ) . '</dt><dd>' . $value . '</dd></div>';
		}
	}

	$html .= '<div class="bqb-fiche__grid' . ( trim( wp_strip_all_tags( $notice_html ) ) ? '' : ' is-single' ) . '">';

	if ( trim( wp_strip_all_tags( $notice_html ) ) ) {
		$html .= '<div class="bqb-fiche__notice"><h2 class="bqb-fiche__h2">Contexte et présentation</h2>' . $notice_html . '</div>';
	}

	$html .= '<aside class="bqb-fiche__infos"><h2 class="bqb-fiche__h2">Informations</h2><dl>' . $dl . '</dl></aside>';
	$html .= '</div>';

	$related = bqb_related_ids( $post_id );
	if ( $related ) {
		$html .= '<section class="bqb-fiche__related"><h2 class="bqb-fiche__h2">Documents associés</h2><div class="bqb-docs__list">';
		foreach ( $related as $id ) {
			$html .= bqb_render_doc_row( $id );
		}
		$html .= '</div></section>';
	}

	$html .= '</div>';

	return $html;
}

add_filter( 'the_content', 'bqb_filter_content', 20 );
function bqb_filter_content( $content ) {
	if ( ! is_singular( BQB_CPT ) || ! in_the_loop() || ! is_main_query() ) {
		return $content;
	}

	if ( ! bqb_settings()['fiche_auto'] ) {
		return $content;
	}

	return bqb_render_fiche( get_the_ID(), $content );
}

function bqb_sc_fiche( $atts ) {
	$atts = shortcode_atts( array( 'id' => 0 ), $atts, 'bqp_document_fiche' );
	$id   = $atts['id'] ? absint( $atts['id'] ) : get_the_ID();

	if ( ! $id || BQB_CPT !== get_post_type( $id ) ) {
		return '';
	}

	return bqb_render_fiche( $id, wpautop( do_shortcode( get_post_field( 'post_content', $id ) ) ) );
}

add_action( 'template_redirect', 'bqb_archive_router', 5 );
function bqb_archive_router() {
	$public = array( 'bqb_collection', 'bqb_theme', 'bqb_personne', 'bqb_organisation' );

	if ( ! is_tax( $public ) ) {
		return;
	}

	$term = get_queried_object();
	if ( ! $term instanceof WP_Term ) {
		return;
	}

	// Une page dédiée existe : l'adresse automatique y renvoie, filtres compris.
	$page = (int) get_term_meta( $term->term_id, 'bqb_page', true );
	if ( $page && 'publish' === get_post_status( $page ) ) {
		$keep = array_intersect_key( wp_unslash( $_GET ), array_flip( array_map( function ( $k ) { return 'f_' . $k; }, array_merge( array_keys( bqb_short_map() ), array( 'q', 'annee', 'tri', 'groupe' ) ) ) ) ); // phpcs:ignore WordPress.Security.NonceVerification
		wp_safe_redirect( add_query_arg( array_map( 'rawurlencode', array_map( 'sanitize_text_field', $keep ) ), get_permalink( $page ) ), 301 );
		exit;
	}

	if ( ! bqb_settings()['archives_auto'] ) {
		return;
	}

	get_header();
	echo '<main id="content" class="site-main bqb-archive">' . bqb_render_archive( $term ) . '</main>'; // phpcs:ignore WordPress.Security.EscapeOutput.OutputNotEscaped
	get_footer();
	exit;
}

function bqb_render_archive( $term ) {
	$labels = array(
		'bqb_collection'   => 'Collection',
		'bqb_theme'        => 'Thème',
		'bqb_personne'     => 'Auteur',
		'bqb_organisation' => 'Organisation',
	);

	$map   = bqb_term_children_map( $term->taxonomy );
	$color = '';

	if ( 'bqb_collection' === $term->taxonomy ) {
		$root  = $term;
		$anc   = get_ancestors( $term->term_id, $term->taxonomy, 'taxonomy' );
		$root  = $anc ? get_term( end( $anc ), $term->taxonomy ) : $term;
		$color = get_term_meta( $root->term_id, 'bqb_couleur', true );
	}

	wp_enqueue_style( 'bqb-front' );

	$html  = '<div class="bqb-page"' . ( $color ? ' style="--bqb-accent:' . esc_attr( $color ) . ';"' : '' ) . '>';
	$html .= bqb_breadcrumb( bqb_term_trail( $term ) );
	$html .= '<header class="bqb-hero">';

	$visual = '';
	$extra  = '';

	if ( 'bqb_personne' === $term->taxonomy ) {
		$photo = (int) get_term_meta( $term->term_id, 'bqb_photo', true );
		$roles = array_intersect_key( bqb_person_roles(), array_flip( (array) get_term_meta( $term->term_id, 'bqb_roles', true ) ) );
		$year  = get_term_meta( $term->term_id, 'bqb_annee_prix', true );
		$prix  = (int) get_term_meta( $term->term_id, 'bqb_prix_id', true );
		$prix  = $prix ? get_term( $prix, 'bqb_prix' ) : null;
		$site  = get_term_meta( $term->term_id, 'bqb_site', true );
		$bio   = get_term_meta( $term->term_id, 'bqb_bio', true );

		$visual = $photo ? wp_get_attachment_image( $photo, 'medium', false, array( 'class' => 'bqb-hero__photo', 'alt' => 'Portrait de ' . $term->name ) ) : '';
		$labels['bqb_personne'] = $roles ? implode( ', ', $roles ) : 'Auteur';

		if ( $prix && ! is_wp_error( $prix ) ) {
			$extra .= '<p class="bqb-hero__prize">' . esc_html( $prix->name ) . ( $year ? ' ' . (int) $year : '' ) . '</p>';
		}
		if ( $bio ) {
			$extra .= '<p class="bqb-hero__text">' . nl2br( esc_html( $bio ) ) . '</p>';
		}
		if ( $site ) {
			$extra .= '<p><a class="bqb-btn bqb-btn--ghost" href="' . esc_url( $site ) . '" target="_blank" rel="noopener">Page personnelle</a></p>';
		}
	}

	if ( 'bqb_organisation' === $term->taxonomy ) {
		$logo  = (int) get_term_meta( $term->term_id, 'bqb_logo', true );
		$type  = get_term_meta( $term->term_id, 'bqb_type_orga', true );
		$types = bqb_org_types();
		$site  = get_term_meta( $term->term_id, 'bqb_site', true );

		$visual = $logo ? wp_get_attachment_image( $logo, 'medium', false, array( 'class' => 'bqb-hero__logo', 'alt' => 'Logo ' . $term->name ) ) : '';

		if ( $type && isset( $types[ $type ] ) ) {
			$labels['bqb_organisation'] = $types[ $type ];
		}
		if ( $site ) {
			$extra .= '<p><a class="bqb-btn bqb-btn--ghost" href="' . esc_url( $site ) . '" target="_blank" rel="noopener">Site internet</a></p>';
		}
	}

	if ( $visual ) {
		$html .= '<div class="bqb-hero__visual">' . $visual . '</div>';
	}

	$html .= '<div class="bqb-hero__body">';
	$html .= '<span class="bqb-hero__eyebrow">' . esc_html( $labels[ $term->taxonomy ] ) . '</span>';
	$html .= '<h1 class="bqb-hero__title">' . esc_html( $term->name ) . '</h1>';

	if ( $term->description ) {
		$html .= '<p class="bqb-hero__text">' . esc_html( $term->description ) . '</p>';
	}

	$html .= $extra . '</div></header>';

	// Sous-catégories, en raccourcis.
	if ( ! empty( $map[ $term->term_id ] ) ) {
		$chips = '';
		foreach ( $map[ $term->term_id ] as $child ) {
			if ( $child->count || bqb_settings()['afficher_vides'] ) {
				$chips .= '<a class="bqb-chip bqb-chip--nav" href="' . esc_url( bqb_term_url( $child ) ) . '">' . esc_html( $child->name ) . ' <span>' . (int) $child->count . '</span></a>';
			}
		}
		if ( $chips ) {
			$html .= '<nav class="bqb-subnav" aria-label="Sous-catégories">' . $chips . '</nav>';
		}
	}

	$atts            = array_fill_keys( array_keys( bqb_short_map() ), '' );
	$atts['annee']   = '';
	$atts['acces']   = '';
	$atts['groupe']  = '';
	$atts['tri']     = '';
	$atts['nombre']  = '';
	$atts['filtres'] = 'oui';
	$atts['titre']   = '';
	$atts['vide']    = 'Aucun document dans cette rubrique pour le moment.';
	$atts['pagination'] = 'oui';

	$html .= bqb_render_documents( $atts, array( bqb_short_name( $term->taxonomy ) => array( $term->slug ) ) );
	$html .= '</div>';

	return $html;
}

/* -------------------------------------------------------------------------
 * 28. [bqp_personnes] : index des personnes ou des organisations
 * ---------------------------------------------------------------------- */

function bqb_sc_personnes( $atts ) {
	$atts = shortcode_atts( array( 'famille' => 'personnes', 'role' => '', 'type' => '', 'groupe' => '' ), $atts, 'bqp_personnes' );
	$orgs = ( 'organisations' === strtolower( $atts['famille'] ) );
	$tax  = $orgs ? 'bqb_organisation' : 'bqb_personne';

	$filter_key = $orgs ? 'type' : 'role';
	$choices    = $orgs ? bqb_org_types() : bqb_person_roles();
	$current    = $atts[ $filter_key ];

	if ( '' === $current && isset( $_GET[ $filter_key ] ) ) { // phpcs:ignore WordPress.Security.NonceVerification
		$current = sanitize_key( wp_unslash( $_GET[ $filter_key ] ) ); // phpcs:ignore WordPress.Security.NonceVerification
	}
	$current = array_key_exists( $current, $choices ) ? $current : '';

	$args = array( 'taxonomy' => $tax, 'hide_empty' => false, 'orderby' => 'name' );

	if ( $current ) {
		$args['meta_query'] = $orgs
			? array( array( 'key' => 'bqb_type_orga', 'value' => $current ) )
			: array( array( 'key' => 'bqb_roles', 'value' => '"' . $current . '"', 'compare' => 'LIKE' ) );
	}

	$terms = get_terms( $args );
	$terms = is_wp_error( $terms ) ? array() : $terms;

	// Sans filtre, on n'affiche que les fiches reliées à au moins un document,
	// ou les lauréats, qui ont leur place même sans document en ligne.
	if ( ! $current ) {
		$terms = array_values(
			array_filter(
				$terms,
				function ( $t ) use ( $orgs ) {
					return $t->count || ( ! $orgs && in_array( 'laureat', (array) get_term_meta( $t->term_id, 'bqb_roles', true ), true ) );
				}
			)
		);
	}

	wp_enqueue_style( 'bqb-front' );

	$html  = '<section class="bqb-people">';
	$html .= '<nav class="bqb-people__filters">';
	$html .= '<a class="bqb-chip bqb-chip--nav' . ( $current ? '' : ' is-on' ) . '" href="' . esc_url( remove_query_arg( $filter_key ) ) . '">Tous</a>';
	foreach ( $choices as $key => $label ) {
		$html .= '<a class="bqb-chip bqb-chip--nav' . ( $current === $key ? ' is-on' : '' ) . '" href="' . esc_url( add_query_arg( $filter_key, $key ) ) . '">' . esc_html( $label ) . '</a>';
	}
	$html .= '</nav>';

	if ( ! $terms ) {
		return $html . '<p class="bqb-docs__empty">Aucune fiche pour le moment.</p></section>';
	}

	$groupe  = $atts['groupe'] ? $atts['groupe'] : ( ( ! $orgs && 'laureat' === $current ) ? 'annee' : 'alpha' );
	$buckets = array();

	foreach ( $terms as $t ) {
		if ( 'annee' === $groupe ) {
			$year = get_term_meta( $t->term_id, 'bqb_annee_prix', true );
			$key  = $year ? 'Promotion ' . (int) $year : 'Année non renseignée';
		} else {
			$key = strtoupper( substr( remove_accents( $t->name ), 0, 1 ) );
		}
		$buckets[ $key ][] = $t;
	}

	'annee' === $groupe ? krsort( $buckets ) : ksort( $buckets );

	foreach ( $buckets as $key => $list ) {
		$html .= '<div class="bqb-group"><h3 class="bqb-group__title">' . esc_html( $key ) . '</h3><div class="bqb-people__grid">';

		foreach ( $list as $t ) {
			$img_id = (int) get_term_meta( $t->term_id, $orgs ? 'bqb_logo' : 'bqb_photo', true );
			$sub    = array();

			if ( $orgs ) {
				$type = get_term_meta( $t->term_id, 'bqb_type_orga', true );
				if ( isset( $choices[ $type ] ) ) {
					$sub[] = $choices[ $type ];
				}
			} else {
				$sub = array_values( array_intersect_key( $choices, array_flip( (array) get_term_meta( $t->term_id, 'bqb_roles', true ) ) ) );
			}

			$initials = '';
			foreach ( preg_split( '/[\s\-]+/u', $t->name ) as $part ) {
				$initials .= function_exists( 'mb_substr' ) ? mb_substr( $part, 0, 1 ) : substr( $part, 0, 1 );
				if ( strlen( $initials ) >= 2 ) {
					break;
				}
			}

			$html .= '<a class="bqb-person' . ( $orgs ? ' is-org' : '' ) . '" href="' . esc_url( bqb_term_url( $t ) ) . '">';
			$html .= '<span class="bqb-person__visual">' . ( $img_id ? wp_get_attachment_image( $img_id, 'thumbnail', false, array( 'alt' => '' ) ) : '<span class="bqb-person__initials">' . esc_html( strtoupper( $initials ) ) . '</span>' ) . '</span>';
			$html .= '<span class="bqb-person__body"><strong>' . esc_html( $t->name ) . '</strong>';
			if ( $sub ) {
				$html .= '<em>' . esc_html( implode( ', ', $sub ) ) . '</em>';
			}
			$html .= '<span class="bqb-person__count">' . sprintf( _n( '%s document', '%s documents', (int) $t->count ), (int) $t->count ) . '</span></span>';
			$html .= '</a>';
		}

		$html .= '</div></div>';
	}

	return $html . '</section>';
}

add_action( 'wp_enqueue_scripts', 'bqb_front_assets' );
function bqb_front_assets() {
	wp_register_style( 'bqb-front', false, array(), BQB_VERSION );
	wp_add_inline_style( 'bqb-front', bqb_front_css() );

	wp_register_script( 'bqb-front', false, array(), BQB_VERSION, true );
	wp_add_inline_script( 'bqb-front', bqb_front_js() );

	$needed = is_singular( BQB_CPT ) || is_tax( array( 'bqb_collection', 'bqb_theme', 'bqb_personne', 'bqb_organisation' ) );

	if ( ! $needed && is_singular() ) {
		$post    = get_post();
		$content = $post ? (string) $post->post_content . (string) get_post_meta( $post->ID, '_elementor_data', true ) : '';
		$needed  = ( false !== strpos( $content, '[bqp_' ) );
	}

	// Chargé dans l'en-tête dès que la page en a besoin, pour éviter tout
	// affichage sans style le temps du chargement.
	if ( $needed ) {
		wp_enqueue_style( 'bqb-front' );
		wp_enqueue_script( 'bqb-front' );
	}
}

function bqb_front_js() {
	return "
	(function(){
		function boot(){
			document.addEventListener('click', function(e){
				var btn = e.target.closest('[data-cite]');
				if (!btn) { return; }
				var text = btn.getAttribute('data-cite');
				var done = function(){ var old = btn.textContent; btn.textContent = 'Référence copiée'; setTimeout(function(){ btn.textContent = old; }, 1800); };
				if (navigator.clipboard) { navigator.clipboard.writeText(text).then(done, function(){ window.prompt('Référence à copier :', text); }); }
				else { window.prompt('Référence à copier :', text); }
			});

			// Adresses de recherche propres : les filtres vides ne sont pas envoyés.
			Array.prototype.forEach.call(document.querySelectorAll('.bqb-filters, .bqb-libnav__search'), function(form){
				form.addEventListener('submit', function(){
					Array.prototype.forEach.call(form.querySelectorAll('input, select'), function(el){
						if (!el.value) { el.disabled = true; }
					});
				});
			});
		}

		if ('loading' === document.readyState) { document.addEventListener('DOMContentLoaded', boot); } else { boot(); }
	})();
	";
}

	function post(data){
		var fd = new FormData();
		fd.append('nonce', CFG.nonce_terms);
		Object.keys(data).forEach(function(k){
			var v = data[k];
			if (Array.isArray(v)) { v.forEach(function(x){ fd.append(k + '[]', x); }); } else { fd.append(k, v); }
		});

	}

	function get(params){
		params.nonce = CFG.nonce_terms;
		var q = Object.keys(params).map(function(k){ return encodeURIComponent(k) + '=' + encodeURIComponent(params[k]); }).join('&');

		else {
			var t = document.createElement('textarea'); t.value = text; document.body.appendChild(t); t.select();
			try { document.execCommand('copy'); done(); } catch (e) { window.prompt('Copiez ce texte :', text); }
			t.remove();
		}
	}

	function message(r, fallback){ return (r && r.data && r.data.message) ? r.data.message : fallback; }

			var radio = e.target.closest('.bqb-radio__opt input');
			if (radio) {
				Array.prototype.forEach.call(document.querySelectorAll('.bqb-radio__opt'), function(o){ o.classList.toggle('is-on', o.querySelector('input').checked); });
			}
		});

		document.addEventListener('click', function(e){
			var b = e.target.closest('[data-copy]');
			if (b) { e.preventDefault(); copy(b.getAttribute('data-copy'), b); }
		});
	}

	function refreshCount(block){
		var n = block.querySelectorAll('.bqb-pill input:checked, .bqb-token').length;
		var c = block.querySelector('[data-count]');
		if (c) { c.textContent = n ? n : ''; }
	}

	/* ------------------------------------------------------------------
	 * Panneau Classement de la fiche document
	 * ------------------------------------------------------------------ */
	function initClassement(){
		var root = document.querySelector('.bqb-class');
		if (!root) { return; }

		/* Filtre des thèmes */
		Array.prototype.forEach.call(root.querySelectorAll('[data-filter]'), function(input){
			var block = input.closest('[data-tax]');
			input.addEventListener('input', function(){
				var q = input.value.trim().toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '');
				Array.prototype.forEach.call(block.querySelectorAll('[data-group]'), function(group){
					var any = false;
					Array.prototype.forEach.call(group.querySelectorAll('.bqb-pill'), function(p){
						var hit = !q || (p.getAttribute('data-name') || '').indexOf(q) !== -1;
						p.classList.toggle('is-hidden', !hit && !p.classList.contains('bqb-pill--root'));
						if (hit) { any = true; }
					});
					group.classList.toggle('is-hidden', !any);
				});
			});
		});

		/* Ajout d'une catégorie depuis la fiche */
		root.addEventListener('click', function(e){
			var open = e.target.closest('[data-add-open]');
			var save = e.target.closest('[data-add-save]');

			if (open) {
				var form = open.parentNode.querySelector('[data-add-form]');
				form.hidden = !form.hidden;
				if (!form.hidden) { form.querySelector('input').focus(); }
				return;
			}

			if (save) { createFromBlock(save.closest('[data-tax]')); }
		});

		root.addEventListener('keydown', function(e){
			if ('Enter' === e.key && e.target.matches('[data-add-input]')) { e.preventDefault(); createFromBlock(e.target.closest('[data-tax]')); }
		});

		function createFromBlock(block){
			var input = block.querySelector('[data-add-input]');
			var name  = input.value.trim();
			var tax   = block.getAttribute('data-tax');
			if (!name) { return; }

			post({ action: 'bqb_term_create', tax: tax, name: name }).then(function(r){
				if (!r || !r.success) { window.alert(message(r, 'Création impossible.')); return; }

				var existing = block.querySelector('.bqb-pill input[value="' + r.data.id + '"]');
				if (existing) {
					existing.checked = true;
					existing.closest('.bqb-pill').classList.add('is-on');
				} else {
					var label = document.createElement('label');
					label.className = 'bqb-pill is-on';
					label.setAttribute('data-name', r.data.name.toLowerCase());
					label.innerHTML = '<input type="checkbox" name="bqb_terms[' + tax + '][]" value="' + parseInt(r.data.id, 10) + '" checked /><span>' + esc(r.data.name) + '</span>';

					var holder = block.querySelector('[data-pills]');
					if (!holder) {
						holder = document.createElement('div');
						holder.className = 'bqb-tree-pick__group';
						holder.setAttribute('data-group', '');
						var wrap = document.createElement('div');
						wrap.className = 'bqb-tree-pick__root';
						wrap.appendChild(label.cloneNode(true));
						holder.appendChild(wrap);
						block.querySelector('.bqb-tree-pick').appendChild(holder);
					} else {
						holder.appendChild(label);
					}
				}

				input.value = '';
				block.querySelector('[data-add-form]').hidden = true;
				refreshCount(block);
			});
		}

		/* Personnes, organisations, mots-clés : recherche et création */
		Array.prototype.forEach.call(root.querySelectorAll('[data-mode="tokens"]'), function(block){
			var tax   = block.getAttribute('data-tax');
			var list  = block.querySelector('[data-token-list]');
			var input = block.querySelector('[data-token-input]');
			var menu  = block.querySelector('[data-token-menu]');
			var timer = null;
			var items = [];
			var active = -1;

			function chosenIds(){ return Array.prototype.map.call(list.querySelectorAll('input'), function(i){ return i.value; }); }

			function add(id, name){
				if (chosenIds().indexOf(String(id)) !== -1) { return; }
				var li = document.createElement('li');
				li.className = 'bqb-token';
				li.innerHTML = '<input type="hidden" name="bqb_terms[' + tax + '][]" value="' + parseInt(id, 10) + '" /><span>' + esc(name) + '</span><button type="button" aria-label="Retirer">×</button>';
				list.appendChild(li);
				refreshCount(block);
			}

			function close(){ menu.hidden = true; items = []; active = -1; }

			function render(q, results){
				menu.innerHTML = '';
				items = [];
				var exact = false;

				results.forEach(function(t){
					if (t.name.toLowerCase() === q.toLowerCase()) { exact = true; }
					var b = document.createElement('button');
					b.type = 'button';
					b.innerHTML = esc(t.name);
					b.addEventListener('click', function(){ add(t.id, t.name); input.value = ''; close(); input.focus(); });
					menu.appendChild(b);
					items.push(b);
				});

				if (!exact) {
					var c = document.createElement('button');
					c.type = 'button';
					c.className = 'is-create';
					c.innerHTML = '+ Créer « ' + esc(q) + ' »';
					c.addEventListener('click', function(){
						post({ action: 'bqb_term_create', tax: tax, name: q }).then(function(r){
							if (!r || !r.success) { window.alert(message(r, 'Création impossible.')); return; }
							add(r.data.id, r.data.name); input.value = ''; close(); input.focus();
						});
					});
					menu.appendChild(c);
					items.push(c);
				}

				active = items.length ? 0 : -1;
				highlight();
				menu.hidden = false;
			}

			function highlight(){ items.forEach(function(b, i){ b.classList.toggle('is-active', i === active); }); }

			input.addEventListener('input', function(){
				clearTimeout(timer);
				var q = input.value.trim();
				if (!q) { close(); return; }
				timer = setTimeout(function(){
					get({ action: 'bqb_term_search', tax: tax, q: q }).then(function(r){ render(q, (r && r.success) ? r.data : []); });
				}, 200);
			});

			input.addEventListener('keydown', function(e){
				if (menu.hidden) { if ('Enter' === e.key) { e.preventDefault(); } return; }
				if ('ArrowDown' === e.key) { e.preventDefault(); active = Math.min(items.length - 1, active + 1); highlight(); }
				else if ('ArrowUp' === e.key) { e.preventDefault(); active = Math.max(0, active - 1); highlight(); }
				else if ('Enter' === e.key) { e.preventDefault(); if (items[active]) { items[active].click(); } }
				else if ('Escape' === e.key) { close(); }
			});

			list.addEventListener('click', function(e){
				if (e.target.closest('button')) { e.target.closest('li').remove(); refreshCount(block); }
			});

			document.addEventListener('click', function(e){ if (!block.contains(e.target)) { close(); } });
		});
	}

	/* ------------------------------------------------------------------
	 * Arborescence
	 * ------------------------------------------------------------------ */
	function initArbo(){
		var box = document.querySelector('[data-arbo]');
		if (!box) { return; }

		var tax = box.getAttribute('data-tax');
		var dragged = null;

		function node(li){ return li.querySelector(':scope > [data-node]'); }

		function startRename(li){
			var n = node(li);
			var name = n.querySelector('[data-name]');
			var input = n.querySelector('[data-rename]');
			name.hidden = true; input.hidden = false; input.value = name.textContent; input.focus(); input.select();

			var finish = function(save){
				input.removeEventListener('blur', onBlur);
				input.removeEventListener('keydown', onKey);
				var val = input.value.trim();
				input.hidden = true; name.hidden = false;
				if (!save || !val || val === name.textContent) { return; }
				name.textContent = val;
				post({ action: 'bqb_arbo', op: 'rename', tax: tax, id: li.getAttribute('data-id'), name: val }).then(function(r){
					if (!r || !r.success) { window.alert(message(r, 'Renommage impossible.')); window.location.reload(); }
				});
			};
			var onBlur = function(){ finish(true); };
			var onKey = function(e){
				if ('Enter' === e.key) { e.preventDefault(); finish(true); }
				if ('Escape' === e.key) { e.preventDefault(); finish(false); }
			};
			input.addEventListener('blur', onBlur);
			input.addEventListener('keydown', onKey);
		}

		function insertHtml(ul, html){
			var t = document.createElement('div');
			t.innerHTML = html;
			var li = t.firstElementChild;
			ul.appendChild(li);
			var empty = box.querySelector('.bqb-arbo__empty');
			if (empty) { empty.remove(); }
			return li;
		}

		/* Ajout au premier niveau */
		var rootInput = box.querySelector('[data-new-root]');
		function addRoot(){
			var name = rootInput.value.trim();
			if (!name) { rootInput.focus(); return; }
			post({ action: 'bqb_arbo', op: 'add', tax: tax, name: name, parent: 0 }).then(function(r){
				if (!r || !r.success) { window.alert(message(r, 'Ajout impossible.')); return; }
				var li = insertHtml(box.querySelector('.bqb-arbo__root'), r.data.html);
				rootInput.value = '';
				li.scrollIntoView({ behavior: 'smooth', block: 'center' });
			});
		}
		box.querySelector('[data-add-root]').addEventListener('click', addRoot);
		rootInput.addEventListener('keydown', function(e){ if ('Enter' === e.key) { e.preventDefault(); addRoot(); } });

		box.addEventListener('click', function(e){
			var nameEl = e.target.closest('[data-name]');
			if (nameEl) { startRename(nameEl.closest('.bqb-arbo__item')); return; }

			var act = e.target.closest('[data-act]');
			if (!act) {
				var close = e.target.closest('[data-panel-close]');
				if (close) { close.closest('[data-panel]').hidden = true; }
				return;
			}

			var li = act.closest('.bqb-arbo__item');
			var id = li.getAttribute('data-id');
			var what = act.getAttribute('data-act');

			if ('rename' === what) { startRename(li); }

			if ('child' === what) {
				var ul = li.querySelector(':scope > .bqb-arbo__list');
				var row = li.querySelector(':scope > .bqb-arbo__newchild');
				if (row) { row.querySelector('input').focus(); return; }
				row = document.createElement('div');
				row.className = 'bqb-arbo__newchild';
				row.innerHTML = '<input type="text" class="bqb-admin__input" placeholder="Nom de la sous-catégorie" /><button type="button" class="button button-primary">Ajouter</button><button type="button" class="button">Annuler</button>';
				li.insertBefore(row, ul);
				var input = row.querySelector('input');
				input.focus();
				var go = function(){
					var name = input.value.trim();
					if (!name) { input.focus(); return; }
					post({ action: 'bqb_arbo', op: 'add', tax: tax, name: name, parent: id }).then(function(r){
						if (!r || !r.success) { window.alert(message(r, 'Ajout impossible.')); return; }
						insertHtml(ul, r.data.html);
						input.value = '';
						input.focus();
					});
				};
				row.querySelector('.button-primary').addEventListener('click', go);
				row.querySelector('.button:not(.button-primary)').addEventListener('click', function(){ row.remove(); });
				input.addEventListener('keydown', function(ev){ if ('Enter' === ev.key) { ev.preventDefault(); go(); } if ('Escape' === ev.key) { row.remove(); } });
			}

			if ('delete' === what) {
				var count = parseInt(act.getAttribute('data-count'), 10) || 0;
				var kids = parseInt(act.getAttribute('data-children'), 10) || 0;
				var name = node(li).querySelector('[data-name]').textContent;
				var msg = 'Supprimer « ' + name + ' » ?';
				if (count) { msg += '\n\n' + count + ' document(s) sont rangés ici. Ils ne seront pas supprimés : ils perdront simplement cette étiquette.'; }
				if (kids) { msg += '\n\nSes ' + kids + ' sous-catégorie(s) remonteront d\'un niveau.'; }
				msg += '\n\nLes shortcodes qui utilisent cette catégorie n\'afficheront plus rien.';
				if (!window.confirm(msg)) { return; }

				post({ action: 'bqb_arbo', op: 'delete', tax: tax, id: id }).then(function(r){
					if (!r || !r.success) { window.alert(message(r, 'Suppression impossible.')); return; }
					if (kids) { window.location.reload(); } else { li.remove(); }
				});
			}

			if ('panel' === what) {
				var panel = li.querySelector(':scope > [data-panel]');
				if (panel.innerHTML) { panel.hidden = !panel.hidden; return; }
				post({ action: 'bqb_arbo', op: 'panel', tax: tax, id: id }).then(function(r){
					if (!r || !r.success) { window.alert(message(r, 'Réglages indisponibles.')); return; }
					panel.innerHTML = r.data.html;
					panel.hidden = false;
					var form = panel.querySelector('[data-panel-form]');
					form.addEventListener('submit', function(ev){
						ev.preventDefault();
						var fd = new FormData(form);
						var data = { action: 'bqb_arbo', op: 'save', tax: tax, id: id };
						var arrays = {};
						fd.forEach(function(v, k){
							if (/\[\]$/.test(k)) { var key = k.replace(/\[\]$/, ''); (arrays[key] = arrays[key] || []).push(v); }
							else { data[k] = v; }
						});
						Object.keys(arrays).forEach(function(k){ data[k] = arrays[k]; });
						if (form.querySelector('[name="natures[]"]') && !arrays.natures) { data.natures = []; data['natures_empty'] = 1; }
						if (form.querySelector('[name="vues[]"]') && !arrays.vues) { data.vues = []; }

						var out = panel.querySelector('[data-panel-msg]');
						out.className = 'bqb-panel__msg';
						out.textContent = 'Enregistrement…';

						post(data).then(function(res){
							if (!res || !res.success) { out.classList.add('is-error'); out.textContent = message(res, 'Erreur.'); return; }
							if (res.data.reload) { window.location.reload(); return; }
							out.textContent = 'Enregistré';
							setTimeout(function(){ out.textContent = ''; }, 2000);
						});
					});
				});
			}
		});

		/* Glisser-déposer entre catégories sœurs */
		box.addEventListener('mousedown', function(e){
			var h = e.target.closest('[data-handle]');
			if (h) { h.closest('.bqb-arbo__item').setAttribute('draggable', 'true'); }
		});

		box.addEventListener('dragstart', function(e){
			dragged = e.target.closest('.bqb-arbo__item');
			if (!dragged) { return; }
			e.dataTransfer.effectAllowed = 'move';
			try { e.dataTransfer.setData('text/plain', dragged.getAttribute('data-id')); } catch (err) {}
			setTimeout(function(){ dragged.classList.add('is-dragging'); }, 0);
		});

		function clearMarks(){
			Array.prototype.forEach.call(box.querySelectorAll('.bqb-drop-before, .bqb-drop-after'), function(el){ el.classList.remove('bqb-drop-before', 'bqb-drop-after'); });
		}

		box.addEventListener('dragover', function(e){
			if (!dragged) { return; }
			var target = e.target.closest('.bqb-arbo__item');
			if (!target || target === dragged || target.parentNode !== dragged.parentNode) { return; }
			e.preventDefault();
			clearMarks();
			var rect = node(target).getBoundingClientRect();
			target.classList.add(e.clientY < rect.top + rect.height / 2 ? 'bqb-drop-before' : 'bqb-drop-after');
		});

		box.addEventListener('drop', function(e){
			if (!dragged) { return; }
			var target = e.target.closest('.bqb-arbo__item');
			if (!target || target === dragged || target.parentNode !== dragged.parentNode) { return; }
			e.preventDefault();
			var before = target.classList.contains('bqb-drop-before');
			clearMarks();
			target.parentNode.insertBefore(dragged, before ? target : target.nextSibling);

			var ids = Array.prototype.map.call(dragged.parentNode.children, function(li){ return li.getAttribute('data-id'); });
			post({ action: 'bqb_arbo', op: 'reorder', tax: tax, ids: ids });
		});

		box.addEventListener('dragend', function(){
			if (dragged) { dragged.classList.remove('is-dragging'); dragged.removeAttribute('draggable'); }
			clearMarks();
			dragged = null;
		});

		/* Recherche */
		var search = box.querySelector('[data-search]');
		search.addEventListener('input', function(){
			var q = search.value.trim().toLowerCase();
			var all = box.querySelectorAll('.bqb-arbo__item');
			Array.prototype.forEach.call(all, function(li){ li.classList.toggle('is-hidden', !!q); });
			if (!q) { return; }
			Array.prototype.forEach.call(all, function(li){
				var name = node(li).querySelector('[data-name]').textContent.toLowerCase();
				if (name.indexOf(q) !== -1) {
					var el = li;
					while (el && el.classList && el.classList.contains('bqb-arbo__item')) {
						el.classList.remove('is-hidden');
						el = el.parentNode.closest('.bqb-arbo__item');
					}
					Array.prototype.forEach.call(li.querySelectorAll('.bqb-arbo__item'), function(c){ c.classList.remove('is-hidden'); });
				}
			});
		});
	}

	/* ------------------------------------------------------------------
	 * Générateur de shortcode
	 * ------------------------------------------------------------------ */
	function initGenerator(){
		var form = document.querySelector('[data-gen-form]');
		if (!form) { return; }

		var code  = document.querySelector('[data-gen-code]');
		var count = document.querySelector('[data-gen-count]');
		var btn   = document.querySelector('[data-gen-copy]');
		var timer = null;
		var taxKeys = ['collection', 'nature', 'theme', 'personne', 'organisation', 'secteur', 'pays', 'prix', 'motcle'];

		function build(){
			var parts = ['bqp_documents'];
			var q = { action: 'bqb_gen_count' };

			Array.prototype.forEach.call(form.querySelectorAll('[data-gen]'), function(el){
				var key = el.getAttribute('data-gen');
				var val = ('checkbox' === el.type) ? (el.checked ? el.value : '') : el.value.trim();
				if (!val) { return; }
				parts.push(key + '="' + val.replace(/"/g, '') + '"');
				if (taxKeys.indexOf(key) !== -1 || 'annee' === key || 'acces' === key) { q[key] = val; }
			});

			var text = '[' + parts.join(' ') + ']';
			code.textContent = text;

			clearTimeout(timer);
			count.textContent = '…';
			timer = setTimeout(function(){
				get(q).then(function(r){
					if (!r || !r.success) { count.textContent = ''; return; }
					var n = r.data.count;
					count.innerHTML = n ? '<strong>' + n + '</strong> document' + (n > 1 ? 's' : '') + ' correspondent aujourd\'hui.' : 'Aucun document pour cette combinaison pour l\'instant. Le shortcode se remplira dès qu\'un document correspondra.';
				});
			}, 250);
		}

		form.addEventListener('input', build);
		form.addEventListener('change', build);
		btn.addEventListener('click', function(){ copy(code.textContent, btn); });
		build();
	}

	ready(function(){
		bindCommon();
		initClassement();
		initArbo();
		initGenerator();
	});
})();
